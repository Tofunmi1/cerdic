# A 160 x 100 mm plate, 5 mm thick, with four square corner holes and a raised rim made from two
# offsets of the outline (4 mm and 7 mm in from the edge), 2 mm tall.
plate = profile.polygon@1(points=[[-0.08, -0.05], [0.08, -0.05], [0.08, 0.05], [-0.08, 0.05]])
hole_a = profile.polygon@1(points=[[-0.071, -0.041], [-0.065, -0.041], [-0.065, -0.035], [-0.071, -0.035]])
hole_b = profile.polygon@1(points=[[0.065, -0.041], [0.071, -0.041], [0.071, -0.035], [0.065, -0.035]])
hole_c = profile.polygon@1(points=[[0.065, 0.035], [0.071, 0.035], [0.071, 0.041], [0.065, 0.041]])
hole_d = profile.polygon@1(points=[[-0.071, 0.035], [-0.065, 0.035], [-0.065, 0.041], [-0.071, 0.041]])
drilled = profile.difference@1(plate, hole_a, hole_b, hole_c, hole_d)
base = solid.extrude@1(drilled, height=5mm)
outer_edge = profile.offset@1(plate, delta=-4mm)
inner_edge = profile.offset@1(plate, delta=-7mm)
ring = profile.difference@1(outer_edge, inner_edge)
rim = solid.extrude@1(ring, height=2mm)
rim_up = xform.move@1(rim, by=[0, 0.005, 0])
mount = solid.union@1(base, rim_up)
output mount
