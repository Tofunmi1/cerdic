# A toy car as ONE fused solid with several colours: booleans keep each part's colour.
plan = profile.polygon@1(points=[[-0.9,-0.4],[0.9,-0.4],[1.0,-0.3],[1.0,0.3],[0.9,0.4],[-0.9,0.4],[-1.0,0.3],[-1.0,-0.3]])
plan_round = profile.smooth@1(plan, iterations=2)
hull = solid.extrude@1(plan_round, height=0.26, divisions=1, twist=0, top_scale=[0.92, 0.92])
hull_up = xform.move@1(hull, by=[0, 0.12, 0])
cabin_plan = profile.polygon@1(points=[[-0.45,-0.33],[0.38,-0.33],[0.5,-0.2],[0.5,0.2],[0.38,0.33],[-0.45,0.33],[-0.55,0.2],[-0.55,-0.2]])
cabin_round = profile.smooth@1(cabin_plan, iterations=2)
cabin = solid.extrude@1(cabin_round, height=0.24, divisions=1, twist=0, top_scale=[0.7, 0.8])
cabin_up = xform.move@1(cabin, by=[-0.05, 0.38, 0])
tyre = solid.torus@1(major=0.11, minor=0.05, segments=48, tube_segments=20)
hub = solid.cylinder@1(radius=0.085, height=0.07, segments=32)
tyre_rot = xform.rotate@1(tyre, angles=[90, 0, 0])
hub_rot = xform.rotate@1(hub, angles=[90, 0, 0])
rubber = solid.material@1(tyre_rot, color=[0.05, 0.05, 0.06], roughness=0.9)
chrome = solid.material@1(hub_rot, color=[0.8, 0.82, 0.86], metallic=1, roughness=0.15)
wheel = solid.union@1(rubber, chrome)
wheel_a = xform.move@1(wheel, by=[0.62, 0.11, 0.42])
wheels_front = xform.mirror@1(wheel_a, axis="z")
wheel_b = xform.move@1(wheel, by=[-0.62, 0.11, 0.42])
wheels_back = xform.mirror@1(wheel_b, axis="z")
lamp = solid.sphere@1(radius=0.06, segments=20)
lamp_a = xform.move@1(lamp, by=[0.98, 0.3, 0.22])
lamps = xform.mirror@1(lamp_a, axis="z")
paint = solid.material@1(hull_up, color=[0.85, 0.12, 0.1], metallic=0.5, roughness=0.25)
glass = solid.material@1(cabin_up, color=[0.35, 0.55, 0.7], metallic=0.3, roughness=0.05)
glow = solid.material@1(lamps, color=[1, 0.95, 0.7], roughness=0.2, emissive=[3, 2.7, 1.5])
car = solid.union@1(paint, glass, wheels_front, wheels_back, glow)
output car
