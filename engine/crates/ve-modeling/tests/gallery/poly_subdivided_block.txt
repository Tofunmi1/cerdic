box = solid.box@1(size=[1, 1, 1])
triangles = poly.from_solid@1(box)
mesh = poly.merge_coplanar@1(triangles)
edges = poly.select@1(mesh, element="edge", by="box", box_min=[-1, -0.1, -1], box_max=[1, 0.1, 1])
beveled = poly.bevel_edges@1(mesh, edges, width=0.12, segments=2)
smooth = poly.subdivide@1(beveled, levels=1, boundary="edges_and_corners")
output smooth
