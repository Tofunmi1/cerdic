group reed
  blade = solid.cone@1(height=1.3, radius=0.008, top_radius=0.0, segments=4)
  blade_up = xform.move@1(blade, by=[0, 0.65, 0])
  bent = xform.bend@1(blade_up, angle=0.25)
  ring = inst.radial@1(bent, count=9, radius=0.05)
  blades = inst.realize@1(ring)
  blades_m = solid.material@1(blades, color=[0.34, 0.46, 0.2], roughness=0.8)
  head_s = solid.cylinder@1(radius=0.016, height=0.12, segments=8)
  head_a = xform.move@1(head_s, by=[0.02, 1.02, 0])
  head_m = solid.material@1(head_a, color=[0.32, 0.18, 0.1], roughness=0.8)
  stalk_s = solid.cylinder@1(radius=0.005, height=0.96, segments=5)
  stalk_a = xform.move@1(stalk_s, by=[0.02, 0.48, 0])
  stalk_m = solid.material@1(stalk_a, color=[0.34, 0.46, 0.2], roughness=0.8)
  reeds = solid.union@1(blades_m, head_m, stalk_m)
  swaying = solid.foliage@1(reeds, alpha_cutoff=0, sway=0.8, flutter=0.2, translucency=0.5)
  expose count = ring.count
  expose color = blades_m.color
  expose head_color = head_m.color
  result swaying
end
