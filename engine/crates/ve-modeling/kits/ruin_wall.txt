group ruin_wall
  wall = solid.box@1(size=[4, 2.6, 0.4])
  wall_at = xform.move@1(wall, by=[0, 1.3, 0])
  window = solid.box@1(size=[0.9, 1.1, 1])
  window_at = xform.move@1(window, by=[-0.8, 1.5, 0])
  notch = solid.box@1(size=[1.4, 1.2, 1])
  notch_turned = xform.rotate@1(notch, angles=[0, 0, 28])
  notch_at = xform.move@1(notch_turned, by=[1.4, 2.7, 0])
  cut = solid.difference@1(wall_at, window_at, notch_at)
  mesh = poly.from_solid@1(cut)
  rounded = poly.round_edges@1(mesh, width=0.03, segments=2)
  shape = poly.to_solid@1(rounded)
  surface = solid.material@1(shape, color=[0.62, 0.36, 0.26], roughness=0.9, recipe="brick")
  expose wall = wall.size
  expose color = surface.color
  expose roughness = surface.roughness
  result surface
end
