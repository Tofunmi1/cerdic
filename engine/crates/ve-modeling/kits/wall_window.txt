group wall_window
  wall = solid.box@1(size=[2.4, 2.6, 0.2])
  wall_at = xform.move@1(wall, by=[0, 1.3, 0])
  hole = solid.box@1(size=[1.0, 1.2, 0.4])
  hole_at = xform.move@1(hole, by=[0, 1.35, 0])
  body = solid.difference@1(wall_at, hole_at)
  outer = solid.box@1(size=[1.04, 1.24, 0.14])
  inner = solid.box@1(size=[0.88, 1.08, 0.2])
  rim = solid.difference@1(outer, inner)
  rim_at = xform.move@1(rim, by=[0, 1.35, 0])
  bar_v = solid.box@1(size=[0.04, 1.08, 0.06])
  bar_h = solid.box@1(size=[0.84, 0.04, 0.06])
  bars = solid.union@1(bar_v, bar_h)
  bars_at = xform.move@1(bars, by=[0, 1.35, 0])
  pane = solid.box@1(size=[0.88, 1.08, 0.012])
  glass = xform.move@1(pane, by=[0, 1.35, 0])
  all = solid.union@1(body, rim_at, bars_at, glass)
  surface = solid.material@1(all, color=[0.78, 0.76, 0.70], roughness=0.55, recipe="painted_wall")
  expose wall = wall.size
  expose color = surface.color
  expose roughness = surface.roughness
  result surface
end
