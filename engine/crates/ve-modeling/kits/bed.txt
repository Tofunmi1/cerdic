group bed
  frame = solid.box@1(size=[1.4, 0.2, 2.0])
  frame_at = xform.move@1(frame, by=[0, 0.1, 0])
  mattress = solid.box@1(size=[1.3, 0.18, 1.9])
  mattress_at = xform.move@1(mattress, by=[0, 0.28, 0.05])
  headboard = solid.box@1(size=[1.4, 0.7, 0.08])
  headboard_at = xform.move@1(headboard, by=[0, 0.45, -1.0])
  all = solid.union@1(frame_at, mattress_at, headboard_at)
  centred = xform.move@1(all, by=[0, 0, 0.02])
  surface = solid.material@1(centred, color=[0.58, 0.42, 0.28], roughness=0.7, recipe="fabric")
  expose frame = frame.size
  expose mattress = mattress.size
  expose headboard = headboard.size
  expose color = surface.color
  expose roughness = surface.roughness
  result surface
end
