group chair
  seat = solid.box@1(size=[0.45, 0.05, 0.45])
  seat_at = xform.move@1(seat, by=[0, 0.45, 0])
  leg = solid.box@1(size=[0.04, 0.45, 0.04])
  leg_at = xform.move@1(leg, by=[-0.19, 0.225, -0.19])
  legs = xform.array@1(leg_at, count_x=2, count_z=2, spacing=[0.38, 0, 0.38])
  back = solid.box@1(size=[0.45, 0.52, 0.04])
  back_at = xform.move@1(back, by=[0, 0.72, -0.21])
  all = solid.union@1(seat_at, legs, back_at)
  surface = solid.material@1(all, color=[0.50, 0.33, 0.19], roughness=0.65, recipe="wood")
  expose seat = seat.size
  expose back = back.size
  expose color = surface.color
  expose roughness = surface.roughness
  result surface
end
