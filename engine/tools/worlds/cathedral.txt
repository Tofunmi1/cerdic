# tattered banners: cloth written in SimL (springs from the mesh edges, pull, a draft with gusts), the top row fixed
let banner_src = "sim banner\nparam wind = vec3(0.0, 0.0, 1.5)\nparam gust = 1.0\nparam damping = 0.986\nparam top = 0.0\ninput mesh flag\nstate pos: vec3[flag]\nstate prev: vec3[flag]\nkernel init each i in pos {\n  pos[i] = rest(flag, i)\n  prev[i] = rest(flag, i)\n}\nkernel predict each i in pos {\n  let r = rest(flag, i)\n  if r.y > top - 0.05 {\n    pos[i] = r\n    prev[i] = r\n  } else {\n    let v = (pos[i] - prev[i]) * damping\n    prev[i] = pos[i]\n    let puff = wind * (0.4 + gust * (0.5 + 0.5 * sin(t * 0.9 + r.y * 0.8 + r.z * 1.3 + r.x * 0.7)))\n    pos[i] += v + (vec3(0.0, -9.8, 0.0) + puff * (6.0 - r.y * 0.4)) * dt * dt\n  }\n}\nkernel solve each i in pos {\n  let r = rest(flag, i)\n  let p = pos[i]\n  if r.y <= top - 0.05 {\n    let corr = vec3(0.0)\n    let count = 0.0\n    for j, len in adjacent(flag, i) {\n      let share = select(rest(flag, j).y > top - 0.05, 1.0, 0.5)\n      corr += pull(p, pos[j], len) * share\n      count += 1.0\n    }\n    if count > 0.0 {\n      p += corr * (1.3 / count)\n    }\n  }\n  pos[i] = p\n}\nschedule predict; repeat 24 { solve }\noutput mesh pos\n"
def banner(tag, bx, bz)
  b{tag}_g = poly.grid@1(width=1.5, height=4.4, columns=24, rows=60)
  b{tag}_r = xform.rotate@1(b{tag}_g, angles=[0, 90, 0])
  b{tag}_a = xform.move@1(b{tag}_r, by=[{bx}, 8.4, {bz}])
  b{tag}_p = sim.program@1(b{tag}_a, rate=60, values="top=10.6", source="${banner_src}")
  b{tag}_m = solid.material@1(b{tag}_p, color=[0.55, 0.1, 0.16], roughness=0.85)
  b{tag}_t = solid.texture@1(b{tag}_m, program="h = surface_height(low=6.0, high=10.8); c = colorize(h, gradient=[0:#2a070d, 0.5:#6b1220, 1:#a82a3a]); albedo = c")
  b{tag}_f = solid.finish@1(b{tag}_t, sheen=0.5, translucency=0.2)
  output banner_{tag} = b{tag}_f
end

# a bat colony wheeling in the open vault: boids by three local rules (spatial hash), kept between y 11 and 17.5
bats = sim.program@1(size=0.2, seed=5, rate=30, source="sim bats\nparam radius = 3.0\nparam max_speed = 8.0\nparam cohesion = 0.6\nparam separation = 4.0\nparam align = 2.5\nstate pos: vec3[220]\nstate vel: vec3[220]\nkernel init each i in pos {\n  pos[i] = vec3(rand(i, 1.0) * 10.0 - 5.0, 12.0 + rand(i, 2.0) * 4.0, rand(i, 3.0) * 24.0 - 12.0)\n  vel[i] = normalize(vec3(rand(i, 4.0), rand(i, 5.0), rand(i, 6.0)) - 0.5) * 5.0\n}\nkernel flock each i in pos {\n  let p = pos[i]\n  let v = vel[i]\n  let centre = vec3(0.0)\n  let heading = vec3(0.0)\n  let push = vec3(0.0)\n  let seen = 0.0\n  for j in neighbors(pos, p, radius) cap 12 {\n    if j != i {\n      centre += pos[j]\n      heading += vel[j]\n      let d = p - pos[j]\n      push += d / max(dot(d, d), 0.05)\n      seen += 1.0\n    }\n  }\n  let acc = vec3(0.0)\n  if seen > 0.0 {\n    acc = (centre / seen - p) * cohesion + (heading / seen - v) * align + push * separation\n  }\n  let target = vec3(sin(t * 0.3) * 3.0, 14.0 + sin(t * 0.5) * 1.5, cos(t * 0.17) * 10.0)\n  acc += normalize(target - p) * 3.0\n  if p.y < 11.0 {\n    acc += vec3(0.0, (11.0 - p.y) * 4.0, 0.0)\n  }\n  if p.y > 17.5 {\n    acc += vec3(0.0, (17.5 - p.y) * 4.0, 0.0)\n  }\n  if abs(p.x) > 6.0 {\n    acc += vec3((6.0 - abs(p.x)) * sign(p.x) * 4.0, 0.0, 0.0)\n  }\n  let nv = v + acc * dt\n  let sp = length(nv)\n  nv = nv * min(1.0, max_speed / max(sp, 0.001))\n  nv = nv * max(1.0, 3.5 / max(sp, 0.001))\n  vel[i] = nv\n}\nkernel move each i in pos { pos[i] += vel[i] * dt }\nschedule flock; move\noutput particles pos\n")
bats_m = solid.material@1(bats, color=[0.02, 0.02, 0.025], roughness=0.8)
output bats = bats_m
