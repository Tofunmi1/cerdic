# ---------- starlings: 300 birds by three local rules over the harbour (neighbors with the spatial hash) ----------
flock = sim.program@1(size=0.38, seed=11, rate=30, source="sim murmuration\nparam radius = 7.0\nparam max_speed = 11.0\nparam cohesion = 0.9\nparam separation = 4.0\nparam align = 3.5\nparam roost = 5.0\nstate pos: vec3[300]\nstate vel: vec3[300]\nkernel init each i in pos {\n  pos[i] = vec3(rand(i, 1.0), rand(i, 2.0), rand(i, 3.0)) * 16.0 - 8.0 + vec3(0.0, 22.0, 12.0)\n  vel[i] = normalize(vec3(rand(i, 4.0), rand(i, 5.0), rand(i, 6.0)) - 0.5) * 8.0\n}\nkernel flock each i in pos {\n  let p = pos[i]\n  let v = vel[i]\n  let centre = vec3(0.0)\n  let heading = vec3(0.0)\n  let push = vec3(0.0)\n  let seen = 0.0\n  for j in neighbors(pos, p, radius) cap 16 {\n    if j != i {\n      centre += pos[j]\n      heading += vel[j]\n      let d = p - pos[j]\n      push += d / max(dot(d, d), 0.05)\n      seen += 1.0\n    }\n  }\n  let acc = vec3(0.0)\n  if seen > 0.0 {\n    acc = (centre / seen - p) * cohesion + (heading / seen - v) * align + push * separation\n  }\n  let target = vec3(sin(t * 0.21) * 30.0, 18.0 + sin(t * 0.37) * 5.0, 14.0 + cos(t * 0.17) * 18.0)\n  acc += normalize(target - p) * roost\n  if p.y < 9.0 {\n    acc += vec3(0.0, (9.0 - p.y) * 3.0, 0.0)\n  }\n  let nv = v + acc * dt\n  let sp = length(nv)\n  nv = nv * min(1.0, max_speed / max(sp, 0.001))\n  nv = nv * max(1.0, 6.0 / max(sp, 0.001))\n  vel[i] = nv\n}\nkernel move each i in pos { pos[i] += vel[i] * dt }\nschedule flock; move\noutput particles pos\n")
