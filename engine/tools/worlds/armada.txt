# the sails: a cloth per sail, top edge and lower corners fixed to this ship's hull (bone_point), the rest blown by gusty wind in world space
let sail_src = "sim sail\nparam wind = vec3(6.0, 0.0, 2.2)\nparam gust = 0.6\nparam damping = 0.98\nparam top = 0.0\nparam bot = 0.0\nparam hw = 1.0\ninput mesh sail\ninput bone hull\nstate pos: vec3[sail]\nstate prev: vec3[sail]\nkernel init each i in pos {\n  let w = bone_point(hull, rest(sail, i))\n  pos[i] = w\n  prev[i] = w\n}\nkernel predict each i in pos {\n  let r = rest(sail, i)\n  let pin = r.y > top - 0.05 || (r.y < bot + 0.05 && abs(r.z) > hw - 0.5)\n  let w = bone_point(hull, r)\n  if pin || distance(pos[i], w) > 8.0 {\n    pos[i] = w\n    prev[i] = w\n  } else {\n    let v = (pos[i] - prev[i]) * damping\n    v = v * min(1.0, 0.12 / max(length(v), 0.0001))\n    prev[i] = pos[i]\n    let puff = wind * (1.0 + gust * sin(t * 1.7 + r.y * 0.6 + r.z * 0.5))\n    pos[i] += v + (vec3(0.0, -9.8, 0.0) + puff * 2.4) * dt * dt\n  }\n}\nkernel solve each i in pos {\n  let r = rest(sail, i)\n  let pin = r.y > top - 0.05 || (r.y < bot + 0.05 && abs(r.z) > hw - 0.5)\n  let p = pos[i]\n  if !pin {\n    let corr = vec3(0.0)\n    let count = 0.0\n    for j, len in adjacent(sail, i) {\n      let rj = rest(sail, j)\n      let pj = rj.y > top - 0.05 || (rj.y < bot + 0.05 && abs(rj.z) > hw - 0.5)\n      corr += pull(p, pos[j], len) * select(pj, 1.0, 0.5)\n      count += 1.0\n    }\n    if count > 0.0 {\n      p += corr * (1.3 / count)\n    }\n  }\n  pos[i] = p\n}\nschedule predict; repeat 16 { solve }\noutput mesh pos\n"
def sail(tag, mx, yc, w, h, ytop, ybot, hw)
  s{tag}_g = poly.grid@1(width={w}, height={h}, columns=12, rows=9)
  s{tag}_r = xform.rotate@1(s{tag}_g, angles=[0, 90, 0])
  s{tag}_a = xform.move@1(s{tag}_r, by=[{mx}, {yc}, 0])
  s{tag}_p = sim.program@1(s{tag}_a, rate=60, values="top=${ytop}; bot=${ybot}; hw=${hw}", source="${sail_src}")
  s{tag}_m = solid.material@1(s{tag}_p, color=[0.9, 0.85, 0.72], roughness=0.9)
  output sail_{tag} = s{tag}_m
end
