# trees round the rim: a bark trunk and two crowns of leaves in autumn colours
def tree(tag, x, z, h, lr, lg, lb)
  t{tag}_trunk = solid.cylinder@1(radius={h * 0.05}, height={h * 0.55}, segments=12)
  t{tag}_trunk_up = xform.move@1(t{tag}_trunk, by=[0, {h * 0.27}, 0])
  t{tag}_trunk_m = solid.material@1(t{tag}_trunk_up, color=[0.3, 0.2, 0.13], roughness=0.95)
  t{tag}_crown = solid.sphere@1(radius={h * 0.3}, segments=14)
  t{tag}_crown_up = xform.move@1(t{tag}_crown, by=[0, {h * 0.68}, 0])
  t{tag}_crown2 = solid.sphere@1(radius={h * 0.22}, segments=12)
  t{tag}_crown2_up = xform.move@1(t{tag}_crown2, by=[{h * 0.14}, {h * 0.86}, {h * 0.05}])
  t{tag}_leaves = solid.union@1(t{tag}_crown_up, t{tag}_crown2_up)
  t{tag}_leaves_m = solid.material@1(t{tag}_leaves, color=[{lr}, {lg}, {lb}], roughness=0.9)
  t{tag}_all = solid.union@1(t{tag}_trunk_m, t{tag}_leaves_m)
  t{tag}_at = xform.move@1(t{tag}_all, by=[{x}, 0, {z}])
  output tree{tag} = t{tag}_at
end

# a fighter: a body, a head, shoulders and a cape of the given colour, and what they carry
def fighter(tag, r, g, b, cr, cg, cb, h)
  f{tag}_body = solid.capsule@1(radius=0.28, height={h * 0.72})
  f{tag}_up = xform.move@1(f{tag}_body, by=[0, {h * 0.4}, 0])
  f{tag}_head = solid.sphere@1(radius=0.17, segments=20)
  f{tag}_hup = xform.move@1(f{tag}_head, by=[0, {h * 0.86}, 0])
  f{tag}_sh = solid.box@1(size=[0.62, 0.14, 0.34])
  f{tag}_shup = xform.move@1(f{tag}_sh, by=[0, {h * 0.72}, 0])
  f{tag}_cape = solid.box@1(size=[0.5, {h * 0.55}, 0.07])
  f{tag}_capeup = xform.move@1(f{tag}_cape, by=[0, {h * 0.48}, -0.22])
  f{tag}_u1 = solid.union@1(f{tag}_up, f{tag}_hup)
  f{tag}_u2 = solid.union@1(f{tag}_u1, f{tag}_shup)
  f{tag}_m = solid.material@1(f{tag}_u2, color=[{r}, {g}, {b}], roughness=0.55, metallic=0.25)
  f{tag}_cm = solid.material@1(f{tag}_capeup, color=[{cr}, {cg}, {cb}], roughness=0.8)
  f{tag}_all = solid.union@1(f{tag}_m, f{tag}_cm)
  output {tag} = f{tag}_all
end
