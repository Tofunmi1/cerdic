# ---- a wheel: tyre and rim revolved about the axle (Y here), turned onto X, moved into place; rot is -90 for the left side and 90 for the right ----
def wheel(tag, wx, wy, wz, rot)
  w{tag}_tp = profile.polygon@1(points=[[0.235, -0.1075], [0.30, -0.108], [0.335, -0.09], [0.34, -0.05], [0.341, 0], [0.34, 0.05], [0.335, 0.09], [0.30, 0.108], [0.235, 0.1075], [0.225, 0]])
  w{tag}_tps = profile.smooth@1(w{tag}_tp, iterations=2)
  w{tag}_tire = solid.revolve@1(w{tag}_tps, angle=360deg, segments=72)
  w{tag}_tm = solid.material@1(w{tag}_tire, color=[0.02, 0.02, 0.022], roughness=0.92)
  w{tag}_tf = solid.finish@1(w{tag}_tm, relief=0.4, relief_scale=60, relief_style="ridged", sheen=0.15)
  w{tag}_tr = xform.rotate@1(w{tag}_tf, angles=[0, 0, {rot}])
  w{tag}_ta = xform.move@1(w{tag}_tr, by=[{wx}, {wy}, {wz}])
  output tire_{tag} = w{tag}_ta
  w{tag}_rp = profile.polygon@1(points=[[0.19, -0.085], [0.228, -0.085], [0.228, 0.095], [0.19, 0.095]])
  w{tag}_rim = solid.revolve@1(w{tag}_rp, angle=360deg, segments=64)
  w{tag}_disc = solid.cylinder@1(radius=0.21, height=0.03, segments=48)
  w{tag}_disc_at = xform.move@1(w{tag}_disc, by=[0, 0.07, 0])
  w{tag}_hub = solid.cylinder@1(radius=0.05, height=0.12, segments=24)
  w{tag}_hub_at = xform.move@1(w{tag}_hub, by=[0, 0.075, 0])
  w{tag}_spoke = solid.box@1(size=[0.15, 0.034, 0.05])
  w{tag}_spk = xform.move@1(w{tag}_spoke, by=[0.12, 0.092, 0])
  w{tag}_s1 = xform.rotate@1(w{tag}_spk, angles=[0, 0, 0])
  w{tag}_s2 = xform.rotate@1(w{tag}_spk, angles=[0, 72, 0])
  w{tag}_s3 = xform.rotate@1(w{tag}_spk, angles=[0, 144, 0])
  w{tag}_s4 = xform.rotate@1(w{tag}_spk, angles=[0, 216, 0])
  w{tag}_s5 = xform.rotate@1(w{tag}_spk, angles=[0, 288, 0])
  w{tag}_ru = solid.union@1(w{tag}_rim, w{tag}_disc_at, w{tag}_hub_at, w{tag}_s1, w{tag}_s2, w{tag}_s3, w{tag}_s4, w{tag}_s5)
  w{tag}_rm = solid.material@1(w{tag}_ru, color=[0.78, 0.8, 0.84], metallic=0.95, roughness=0.22)
  w{tag}_rf = solid.finish@1(w{tag}_rm, clearcoat=0.5, clearcoat_roughness=0.1)
  w{tag}_rr = xform.rotate@1(w{tag}_rf, angles=[0, 0, {rot}])
  w{tag}_ra = xform.move@1(w{tag}_rr, by=[{wx}, {wy}, {wz}])
  output rim_{tag} = w{tag}_ra
  w{tag}_bd = solid.cylinder@1(radius=0.16, height=0.03, segments=40)
  w{tag}_bdm = solid.material@1(w{tag}_bd, color=[0.25, 0.25, 0.27], metallic=0.8, roughness=0.5)
  w{tag}_bdr = xform.rotate@1(w{tag}_bdm, angles=[0, 0, {rot}])
  w{tag}_bda = xform.move@1(w{tag}_bdr, by=[{wx}, {wy}, {wz}])
  output brake_{tag} = w{tag}_bda
end
