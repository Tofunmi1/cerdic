# SKIN BODY, HIGH DETAIL: one closed surface for the whole woman, 1.75 m, from a skeleton of about 100 joints (poly.skin, Blender's Skin modifier) with FIVE-digit hands (a thumb and four fingers,
# three phalanges each), a ribcage and spine chain, skull, shaped legs and feet. The skeleton alone gives a clean base of quads; then the body is subdivided, ANATOMY is pushed into it with surface
# programs (muscle and fat masses placed at measured landmark positions: bust, deltoid, trapezius, back, glute, quad, hamstring, calf, shin, knee, biceps, forearm ...), subdivided once more for a smooth
# final surface, and given a very small skin-scale relief. Every pass is a separate named node, so one can be tuned or lifted alone.
# Skeleton rules (see poly.skin): the surface stops at an end joint; bones at a branch need about 90 degrees between them; two branch joints need a bone between them. Hence the palm splits into two
# sub-hubs (pa, pb) with two fingers each, and the toes are one end block. Metres, +y up, facing +z. Left side (+x) and right side (-x) are both written out.

let hand_r = "palm_r -0.272 0.805 0.04 0.032 0.016; pa_r -0.250 0.785 0.044 0.018 0.012; pb_r -0.296 0.785 0.044 0.018 0.012; f1a_r -0.232 0.755 0.046 0.0085 0.0085; f1b_r -0.230 0.715 0.05 0.0075 0.0075; f1c_r -0.230 0.685 0.052 0.0065 0.0065; f2a_r -0.268 0.755 0.046 0.0085 0.0085; f2b_r -0.268 0.71 0.05 0.0075 0.0075; f2c_r -0.268 0.675 0.052 0.0065 0.0065; f3a_r -0.280 0.755 0.046 0.0085 0.0085; f3b_r -0.280 0.71 0.05 0.0075 0.0075; f3c_r -0.280 0.675 0.052 0.0065 0.0065; f4a_r -0.316 0.755 0.046 0.0075 0.0075; f4b_r -0.318 0.72 0.05 0.0065 0.0065; f4c_r -0.318 0.695 0.052 0.0055 0.0055; th1_r -0.230 0.845 0.085 0.016 0.014; th2_r -0.218 0.805 0.10 0.012 0.012; th3_r -0.212 0.775 0.108 0.009 0.009"

let bones_t = "pelvis-spine1; spine1-spine2; spine2-spine3; spine3-chest; chest-chest2; chest2-neck_base; neck_base-neck; neck-skull; skull-crown"
let bones_l = "chest2-shoulder_l; shoulder_l-uarm_l; uarm_l-elbow_l; elbow_l-farm_l; farm_l-wrist_l; wrist_l-palm_l; palm_l-pa_l; palm_l-pb_l; pa_l-f1a_l; f1a_l-f1b_l; f1b_l-f1c_l; pa_l-f2a_l; f2a_l-f2b_l; f2b_l-f2c_l; pb_l-f3a_l; f3a_l-f3b_l; f3b_l-f3c_l; pb_l-f4a_l; f4a_l-f4b_l; f4b_l-f4c_l; wrist_l-th1_l; th1_l-th2_l; th2_l-th3_l; pelvis-hip_l; hip_l-thigh_l; thigh_l-knee_l; knee_l-calf_l; calf_l-shin_l; shin_l-ankle_l; ankle_l-ball_l; ball_l-toes_l"
let bones_r = "chest2-shoulder_r; shoulder_r-uarm_r; uarm_r-elbow_r; elbow_r-farm_r; farm_r-wrist_r; wrist_r-palm_r; palm_r-pa_r; palm_r-pb_r; pa_r-f1a_r; f1a_r-f1b_r; f1b_r-f1c_r; pa_r-f2a_r; f2a_r-f2b_r; f2b_r-f2c_r; pb_r-f3a_r; f3a_r-f3b_r; f3b_r-f3c_r; pb_r-f4a_r; f4a_r-f4b_r; f4b_r-f4c_r; wrist_r-th1_r; th1_r-th2_r; th2_r-th3_r; pelvis-hip_r; hip_r-thigh_r; thigh_r-knee_r; knee_r-calf_r; calf_r-shin_r; shin_r-ankle_r; ankle_r-ball_r; ball_r-toes_r"

base = poly.skin@1(joints="${torso}; ${arm_l}; ${hand_l}; ${leg_l}; ${arm_r}; ${hand_r}; ${leg_r}", bones="${bones_t}; ${bones_l}; ${bones_r}", symmetry="x", branch_smoothing=0.0)
s2 = poly.subdivide@1(base, levels=1)

# ---------------- ANATOMY: masses pushed along the surface normal; each term is amplitude * exp(-(squared distances / 2 sigma^2)) around a landmark; x is mirrored with abs ----------------
# (amplitudes are metres: + builds a muscle or fat mass, - digs a hollow. Coefficients are 1 / (2 sigma^2): 408 = sigma 35 mm, 312 = 40 mm, 247 = 45 mm, 200 = 50 mm, 165 = 55 mm, 139 = 60 mm, 102 = 70 mm, 35 = 120 mm.)
a_torso = poly.program@1(s2, mode="along_normal", code="x = abs(p.x); y = p.y; z = p.z; out = 0.028 * exp(-(247 * (x - 0.085) * (x - 0.085) + 165 * (y - 1.30) * (y - 1.30) + 247 * (z - 0.085) * (z - 0.085))) + 0.012 * exp(-(408 * (x - 0.19) * (x - 0.19) + 408 * (y - 1.39) * (y - 1.39) + 300 * z * z)) + 0.010 * exp(-(139 * (x - 0.09) * (x - 0.09) + 300 * (y - 1.45) * (y - 1.45) + 200 * (z + 0.03) * (z + 0.03))) + 0.008 * exp(-(139 * (x - 0.11) * (x - 0.11) + 60 * (y - 1.25) * (y - 1.25) + 250 * (z + 0.075) * (z + 0.075))) - 0.007 * exp(-(40 * (y - 1.11) * (y - 1.11))) * (1.0 - step(0.2, x)) + 0.007 * exp(-(300 * x * x + 250 * (y - 1.0) * (y - 1.0) + 200 * (z - 0.09) * (z - 0.09)))")
a_hips = poly.program@1(a_torso, mode="along_normal", code="x = abs(p.x); y = p.y; z = p.z; out = 0.030 * exp(-(139 * (x - 0.09) * (x - 0.09) + 102 * (y - 0.93) * (y - 0.93) + 200 * (z + 0.09) * (z + 0.09))) + 0.006 * exp(-(300 * (x - 0.155) * (x - 0.155) + 200 * (y - 0.97) * (y - 0.97) + 300 * z * z)) - 0.004 * exp(-(1500 * x * x + 100 * (y - 1.28) * (y - 1.28))) * smoothstep(0.03, 0.08, z) + 0.008 * exp(-(247 * (x - 0.215) * (x - 0.215) + 312 * (y - 1.22) * (y - 1.22) + 312 * (z - 0.03) * (z - 0.03))) + 0.007 * exp(-(247 * (x - 0.215) * (x - 0.215) + 312 * (y - 1.22) * (y - 1.22) + 312 * (z + 0.03) * (z + 0.03))) + 0.008 * exp(-(312 * (x - 0.25) * (x - 0.25) + 200 * (y - 1.04) * (y - 1.04) + 312 * (z - 0.03) * (z - 0.03)))")
a_legs = poly.program@1(a_hips, mode="along_normal", code="x = abs(p.x); y = p.y; z = p.z; out = 0.022 * exp(-(200 * (x - 0.10) * (x - 0.10) + 35 * (y - 0.72) * (y - 0.72) + 200 * (z - 0.05) * (z - 0.05))) + 0.012 * exp(-(247 * (x - 0.085) * (x - 0.085) + 139 * (y - 0.55) * (y - 0.55) + 247 * (z - 0.045) * (z - 0.045))) + 0.014 * exp(-(200 * (x - 0.10) * (x - 0.10) + 35 * (y - 0.72) * (y - 0.72) + 200 * (z + 0.05) * (z + 0.05))) + 0.022 * exp(-(312 * (x - 0.095) * (x - 0.095) + 102 * (y - 0.38) * (y - 0.38) + 312 * (z + 0.045) * (z + 0.045))) - 0.005 * exp(-(312 * (x - 0.095) * (x - 0.095) + 139 * (y - 0.30) * (y - 0.30) + 312 * (z - 0.04) * (z - 0.04))) + 0.010 * exp(-(312 * (x - 0.098) * (x - 0.098) + 312 * (y - 0.50) * (y - 0.50) + 247 * (z - 0.055) * (z - 0.055))) + 0.012 * exp(-(247 * (x - 0.07) * (x - 0.07) + 35 * (y - 0.72) * (y - 0.72) + 300 * z * z))")
s3 = poly.subdivide@1(a_legs, levels=3)
# the skin's own relief: tiny irregularity a fraction of a millimetre deep, so the surface is never perfectly smooth (a texture program adds the pores and tone later)
skin_detail = poly.program@1(s3, mode="along_normal", code="out = 0.0004 * perlin(p * 220.0, 3) + 0.0002 * perlin(p * 600.0, 5)")
skin_m = solid.material@1(skin_detail, color=[0.84, 0.64, 0.52], roughness=0.55)
skin_f = solid.finish@1(skin_m, subsurface=0.4, clearcoat=0.08, clearcoat_roughness=0.5)
output body_skin = skin_f
