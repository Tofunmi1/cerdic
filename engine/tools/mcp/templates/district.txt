sh5_t = solid.texture@1(sh5_m, program="wall = image(name=\"Concrete034_Color\", repeat_x=3, repeat_y=2); wall_r = image(name=\"Concrete034_Roughness\", repeat_x=3, repeat_y=2); wall_n = image(name=\"Concrete034_Normal\", repeat_x=3, repeat_y=2); shutter = image(name=\"CorrugatedSteel009_Color\", repeat_x=2, repeat_y=1); front = expr(code=\"fx = fract(p.x * 2.0); out = step(0.05, fx) * step(0.05, 1.0 - fx) * step(0.03, p.y) * step(p.y, 0.3)\"); sign = expr(code=\"out = step(0.33, p.y) * step(p.y, 0.44)\"); signc = expr(code=\"out = hsv(fract(floor(p.x * 2.0) * 0.37 + 0.1), 0.6, 0.85)\"); glass = expr(code=\"out = mix(vec3(0.03, 0.05, 0.07), vec3(0.45, 0.4, 0.3), smoothstep(0.05, 0.3, p.y) * 0.5)\"); a1 = blend(signc, wall, sign, blend_type=normal, amount=1.0); albedo = blend(glass, a1, front, blend_type=normal, amount=1.0); roughness = wall_r; normal = wall_n", size="512")
output shop_brick2 = sh5_t
rf5 = city.buildings@1(size=[300, 300], block=40, street_width=9, lot_min=6, lot_max=14, irregularity=0.5, seed=3, part="roof", styles=6, style=5)
rf5_m = solid.material@1(rf5, color=[1, 1, 1], roughness=1.0)
rf5_t = solid.texture@1(rf5_m, program="one = expr(code=\"out = vec3(1.0)\"); tar =image(name=\"Asphalt033_Color\", repeat_x=3, repeat_y=3); tar_r = image(name=\"Asphalt033_Roughness\", repeat_x=3, repeat_y=3); tar_n = image(name=\"Asphalt033_Normal\", repeat_x=3, repeat_y=3); wear = expr(code=\"out = vec3(0.7 + 0.3 * fbm(p * 7.0))\"); albedo = blend(wear, tar, one, blend_type=multiply, amount=1.0); roughness = tar_r; normal = tar_n", size="512")
output roof_brick2 = rf5_t
