# a tapered limb: a tube between two points turned into a solid (h is half the box the solid is built in)
def limb(tag, x0, y0, z0, x1, y1, z1, r0, r1, h, par)
  s{tag}_f = field.tube@1(points=[[{x0}, {y0}, {z0}], [{x1}, {y1}, {z1}]], radius={r0}, end_radius={r1})
  s{tag}_n = field.noise@1(s{tag}_f, amplitude=0.006, scale=0.06, octaves=2)
  s{tag}_s = field.to_solid@1(s{tag}_n, min=[{(x0 + x1) / 2 - h}, {(y0 + y1) / 2 - h}, {(z0 + z1) / 2 - h}], max=[{(x0 + x1) / 2 + h}, {(y0 + y1) / 2 + h}, {(z0 + z1) / 2 + h}], edge_length=0.018)
  s{tag}_m = solid.material@1(s{tag}_s, color=[0.82, 0.57, 0.25], roughness=0.85)
  s{tag}_t = solid.texture@1(s{tag}_m, program="${fur_prog}", tile=0.5, size="256")
  s{tag}_p = anim.parent@1(s{tag}_t, parent="${par}")
end
# a gold armour piece: an ellipsoid shell (radii rx, ry, rz at x, y, z) with a clear coat
def plate(tag, rx, ry, rz, x, y, z, par)
  g{tag}_f = field.ellipsoid@1(radii=[{rx}, {ry}, {rz}])
  g{tag}_m = field.move@1(g{tag}_f, by=[{x}, {y}, {z}])
  g{tag}_n = field.noise@1(g{tag}_m, amplitude=0.004, scale=0.05, octaves=2, mode="ridged")
  g{tag}_s = field.to_solid@1(g{tag}_n, min=[{x - rx - 0.04}, {y - ry - 0.04}, {z - rz - 0.04}], max=[{x + rx + 0.04}, {y + ry + 0.04}, {z + rz + 0.04}], edge_length=0.015)
  g{tag}_c = solid.material@1(g{tag}_s, color=[1.0, 0.8, 0.3], metallic=0.85, roughness=0.24)
  g{tag}_t = solid.texture@1(g{tag}_c, program="${gold_prog}", tile=0.4, size="256")
  g{tag}_x = solid.finish@1(g{tag}_t, clearcoat=0.5, clearcoat_roughness=0.2)
  g{tag}_p = anim.parent@1(g{tag}_x, parent="${par}")
end
