# THE FALLEN TITAN. A dying stone giant in a volcanic canyon at dusk: ridged ground sorted into ash forest, scorched meadow and cliff rock by terrain.biomes,
# a lava sea whose surface is a live animated texture, and the titan itself built from smooth fields, coming apart.
sky = scene.sky@1(sun_elevation=3, sun_azimuth=70, sun_intensity=8, turbidity=6, clouds=false, seed=7, haze=0.07, haze_color=[1.0, 0.4, 0.16], haze_height=60, ground_color=[0.12, 0.06, 0.04])
output sky = sky

land = terrain.tile@1(size=[600, 600], cell=3, seed=21, height=90, scale=420, octaves=5, roughness=0.42, style="hills", basis="simplex")
b = terrain.biomes@1(land, rules="forest: height 14..90, slope 0..26, wet 0.4..1; ash: height 10..120, slope 0..26, wet 0..0.6; cliff: slope 27..", blend=0.5, wet_scale=120, seed=5)
ground = solid.material@1(b, color=[0.16, 0.13, 0.12], roughness=0.97)
output ground = ground

# Charred forest, cinder meadow and cliff boulders, each planted by its own biome layer.
trunk = solid.cone@1(radius=1.1, height=15, top_radius=0.3, segments=7)
crown = solid.cone@1(radius=4.2, height=9, segments=8)
crown_up = xform.move@1(crown, by=[0, 9, 0])
snag = solid.union@1(trunk, crown_up)
snag_m = solid.material@1(snag, color=[0.035, 0.03, 0.028], roughness=0.95)
snags = scatter.place@1(snag_m, b, layer="biome_forest", radius=7, seed=1, max_slope=28, scale=[0.7, 1.7])
output snags = snags
cinder = solid.sphere@1(radius=0.8, segments=8)
cinder_m = solid.material@1(cinder, color=[0.08, 0.05, 0.04], roughness=0.9, emissive=[0.5, 0.1, 0.0])
cinders = scatter.place@1(cinder_m, b, layer="biome_ash", radius=3.2, seed=2, density=0.5, max_slope=30, scale=[0.5, 1.6])
output cinders = cinders
boulder = solid.sphere@1(radius=3, segments=9)
boulder_m = solid.material@1(boulder, color=[0.09, 0.085, 0.085], roughness=0.92)
boulders = scatter.place@1(boulder_m, b, layer="biome_cliff", radius=9, seed=3, sink=0.3, scale=[0.6, 2.2])
output boulders = boulders

lava = solid.box@1(size=[620, 2, 620])
lava_m = solid.material@1(lava, color=[1, 1, 1], roughness=0.35, emissive=[1, 1, 1])
# A texture goes on AFTER the material (a material applied later would replace it).
lava_t = solid.texture@1(lava_m, program="d = expr(code=\"w = fbm(vec2(p.x * 9.0 + t * 0.05, p.y * 9.0)); n = fbm(vec2(p.x * 22.0 + w * 4.0 - t * 0.04, p.y * 22.0 + w * 4.0 + t * 0.03)); k = smoothstep(0.40, 0.58, n); out = vec3(0.025 + 0.06 * k, 0.014 + 0.02 * k, 0.012)\")\nh = expr(code=\"w = fbm(vec2(p.x * 9.0 + t * 0.05, p.y * 9.0)); n = fbm(vec2(p.x * 22.0 + w * 4.0 - t * 0.04, p.y * 22.0 + w * 4.0 + t * 0.03)); m = 1.0 - smoothstep(0.40, 0.58, n); out = vec3(3.0 * m, 0.5 * m * m, 0.02 * m * m * m)\")\nalbedo = d\nemissive = h", size="512")
lava_p = xform.move@1(lava_t, by=[300, 52, 300])
output lava = lava_p

# The dais: a broad stepped drum of dark stone rising out of the lava.

# One part: rocky noise (the same everywhere, so seams match), sampled to a solid, stone material, released at `start` seconds with a sideways
# velocity and a turning rate, falling to the dais (local floor -3), then placed on the dais.
def build(name, src, x0, y0, z0, x1, y1, z1)
  p_{name}_n = field.noise@1(${src}, amplitude=1.1, scale=7, octaves=3, persistence=0.5, seed=4)
  p_{name}_s = field.to_solid@1(p_{name}_n, min=[{x0}, {y0}, {z0}], max=[{x1}, {y1}, {z1}], edge_length=1.0)
  p_{name}_k = solid.material@1(p_{name}_s, color=[1, 1, 1], roughness=0.9, emissive=[1, 1, 1])
  p_{name}_m = solid.texture@1(p_{name}_k, program="a = expr(code=\"n = fbm(vec2(p.x * 11.0, p.y * 11.0)); out = vec3(0.07 + 0.1 * n, 0.065 + 0.08 * n, 0.06 + 0.07 * n)\")\nc = expr(code=\"n = fbm(vec2(p.x * 5.0, p.y * 5.0)); v = 1.0 - smoothstep(0.0, 0.035, abs(n - 0.52)); out = vec3(2.6 * v, 0.55 * v, 0.03 * v)\")\nalbedo = a\nemissive = c", size="512")
end

# A frozen moment of the collapse: the part's own centre is moved to the origin (n is the centre negated), turned by rx ry rz radians, set down at (tx ty tz)
# and placed on the dais.
def pose(name, nx, ny, nz, rx, ry, rz, tx, ty, tz)
  p_{name}_a = xform.move@1(p_{name}_m, by=[{nx}, {ny}, {nz}])
  p_{name}_r = xform.rotate@1(p_{name}_a, angles=[{rx}, {ry}, {rz}])
  p_{name}_b = xform.move@1(p_{name}_r, by=[{tx}, {ty}, {tz}])
  p_{name}_p = xform.move@1(p_{name}_b, by=[300, 114, 300])
  output {name} = p_{name}_p
end
