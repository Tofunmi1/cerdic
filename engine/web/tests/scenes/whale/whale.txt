# a part: a builder call, a material and (optionally) a parent
def wpart(name, call, r, g, b, met, rgh, par)
  {name}_s = poly.script@1(code="${jw} ${jp} ${call}")
  {name}_m = solid.material@1(${name}_s, color=[{r}, {g}, {b}], metallic={met}, roughness={rgh})
  {name}_p = anim.parent@1(${name}_m, parent="${par}")
  output {name} = ${name}_p
end

# the tail: three skinned pieces in a chain, each pitching about its own root a little more and later than the one before; the flukes ride the last piece and flap on their own
def tailpiece(name, joints, bones, piv, amp, ph, par)
  {name}_k = poly.skin@1(joints="${joints}", bones="${bones}", symmetry="none")
  {name}_d = poly.subdivide@1(${name}_k, levels=2)
  {name}_m = solid.material@1(${name}_d, color=[0.72, 0.4, 0.24], metallic=0.85, roughness=0.4)
  {name}_t = solid.texture@1(${name}_m, tile=3, size="512", program="${plates}")
  {name}_w = anim.wave@1(${name}_t, property="rotation", amplitude=[{amp}, 0, 0], period=5, phase={ph}, pivot=[${piv}])
  {name}_p = anim.parent@1(${name}_w, parent="${par}")
  output {name} = ${name}_p
end

# a brass gear: N teeth, its centre (0, y, z), a starting phase (so neighbours mesh), turning ang degrees (a whole number of turns) every 24 s, about its own axis (x)
def gear(tag, N, y, z, ph, ang)
  gr{tag}_s = poly.script@1(code="${jw} ${jp} gear(0, ${y}, ${z}, ${N}, ${ph})")
  gr{tag}_m = solid.material@1(gr${tag}_s, color=[0.9, 0.68, 0.26], metallic=0.9, roughness=0.3)
  gr{tag}_a = anim.track@1(gr${tag}_m, property="rotation", keys="0:0,0,0; 24:${ang},0,0", interpolation="linear", pivot=[0, ${y}, ${z}])
  gr{tag}_p = anim.parent@1(gr${tag}_a, parent="whale_hull")
  output gear_{tag} = gr${tag}_p
end
