table_a = xform.move@1(table_s, by=[0.5, -0.05, 0.25])

output table = table_m

def pane(tag, sx, sy, sz, x, y, z)
  p{tag}_s = solid.box@1(size=[{sx}, {sy}, {sz}])
  p{tag}_a = xform.move@1(p{tag}_s, by=[{x}, {y}, {z}])
  p{tag}_m = solid.material@1(p{tag}_a, color=[0.85, 0.95, 1.0], roughness=0.05, opacity=0.18)
  output glass_{tag} = p{tag}_m
end
