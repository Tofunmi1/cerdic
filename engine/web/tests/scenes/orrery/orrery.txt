# A band (a lathe with a flange), gear teeth on its rim, tilted into its plane, moved to the centre, bronze; a spin about the ring's own axis, carried by the ring outside it.
# Runes are a separate glowing part on the ring.
def ring(k, R, hw, hh, ax, az, spin, par, gk)
  rgp_{k} = profile.polygon@1(points=[[{R - hw}, {0 - hh}], [{R + hw}, {0 - hh}], [{R + hw * 1.3}, {0 - hh * 0.5}], [{R + hw * 1.3}, {hh * 0.5}], [{R + hw}, {hh}], [{R - hw}, {hh}]])
  rgs_{k} = solid.revolve@1(rgp_{k}, angle=360deg, segments=96)
  rtb_{k} = solid.box@1(size=[{hw * 1.6}, 0.3, 0.3])
  rtm_{k} = xform.move@1(rtb_{k}, by=[0, {hh + 0.12}, 0])
  rtr_{k} = xform.radial@1(rtm_{k}, count=36, radius={R})
  rgu_{k} = solid.union@1(rgs_{k}, rtr_{k})
  rgo_{k} = xform.rotate@1(rgu_{k}, angles=[{ax}, 0, {az}])
  rgc_{k} = xform.move@1(rgo_{k}, by=[0, 15, 0])
  rgm_{k} = solid.material@1(rgc_{k}, color=[1, 1, 1], metallic=0.9, roughness=0.4)
  rgt_{k} = solid.texture@1(rgm_{k}, program="${bronze}", tile=0, size="512")
  rgk_{k} = anim.track@1(rgt_{k}, property="rotation", keys={spin}, interpolation="linear", pivot=[0, 15, 0])
  rgp2_{k} = anim.parent@1(rgk_{k}, parent={par})
  output ring{k} = rgp2_{k}
  rub_{k} = solid.box@1(size=[0.1, {hh * 0.9}, 0.34])
  rur_{k} = xform.radial@1(rub_{k}, count=24, radius={R + hw * 1.3 + 0.03})
  ruo_{k} = xform.rotate@1(rur_{k}, angles=[{ax}, 0, {az}])
  ruc_{k} = xform.move@1(ruo_{k}, by=[0, 15, 0])
  rum_{k} = solid.material@1(ruc_{k}, color=[0.4, 0.8, 1.0], roughness=0.3, emissive=[0.8, 2.6, 5.0])
  rug_{k} = anim.track@1(rum_{k}, property="glow", keys={gk}, interpolation="smooth")
  rus_{k} = anim.track@1(rug_{k}, property="rotation", keys={spin}, interpolation="linear", pivot=[0, 15, 0])
  rup_{k} = anim.parent@1(rus_{k}, parent={par})
  output runes{k} = rup_{k}
end
