# A side-scroller: platforms with gaps over a bottomless pit, three coins, a hero that runs and double jumps (game.controller in platformer
# mode, locked to the depth axis), and a side camera with a dead zone and bounds. D/A run, Space jumps (hold for height). Falling below the
# limit puts the hero back on the last safe ground, takes a life and raises "fell"; three falls end the game. Collect all three coins to win.
# Flow: a title (Enter or A), pause (Esc or Start), a win screen and a game-over screen, each left with Enter or A to the title.
sky = scene.sky@1(sun_elevation=30, sun_azimuth=-160, ground_color=[0.2, 0.25, 0.3])
ambient = scene.environment@1(sky_color=[0.55, 0.7, 0.9], ground_color=[0.3, 0.3, 0.32], intensity=0.6)

hero_body = solid.capsule@1(radius=0.25, height=1.1)
hero_up = xform.move@1(hero_body, by=[0, 0.55, 0])
hero_head = solid.sphere@1(radius=0.2, segments=24)
hero_hup = xform.move@1(hero_head, by=[0, 1.3, 0])
hero_scarf = solid.box@1(size=[0.5, 0.12, 0.34])
hero_sup = xform.move@1(hero_scarf, by=[0, 1.05, 0])
hero_u1 = solid.union@1(hero_up, hero_hup)
hero_u2 = solid.union@1(hero_u1, hero_sup)
hero_m = solid.material@1(hero_u2, color=[0.9, 0.45, 0.15], roughness=0.5)
output hero = hero_m

plat = solid.box@1(size=[4, 1, 3])
plat_up = xform.move@1(plat, by=[0, 0, 0])
plat_m = solid.material@1(plat_up, color=[0.5, 0.46, 0.4], roughness=0.85, recipe="concrete")
output plat = plat_m
base = solid.box@1(size=[6, 1, 3])
base_m = solid.material@1(base, color=[0.45, 0.5, 0.4], roughness=0.85, recipe="concrete")
output base = base_m

coin_disc = solid.cylinder@1(radius=0.3, height=0.08, segments=32)
coin_up = xform.rotate@1(coin_disc, angles=[90, 0, 0])
coin_m = solid.material@1(coin_up, color=[1.0, 0.8, 0.2], roughness=0.25, metallic=0.9)
output coin = coin_m

k_hero = game.kind@1(name="player", size=0.3, speed=6, tags=["hero"], part="hero")
k_base = game.kind@1(name="base", size=1, solid=[6, 1, 3], part="base")
k_plat = game.kind@1(name="plat", size=1, solid=[4, 1, 3], part="plat")
k_coin = game.kind@1(name="coin", size=0.3, tags=["coin"], part="coin")

s_hero = game.spawn@1(kind="player", at=[0, 0, 0])
s_p0 = game.spawn@1(kind="base", at=[0, -0.5, 0])
s_p1 = game.spawn@1(kind="plat", at=[7, -0.5, 0])
s_p2 = game.spawn@1(kind="plat", at=[13, 0, 0])
s_p3 = game.spawn@1(kind="plat", at=[19, 0.5, 0])
s_p4 = game.spawn@1(kind="plat", at=[25, 1.0, 0])
c1 = game.spawn@1(kind="coin", at=[7, 0.5, 0])
c2 = game.spawn@1(kind="coin", at=[13, 1.0, 0])
c3 = game.spawn@1(kind="coin", at=[25, 2.0, 0])

score = game.var@1(name="score", value=0)
falls = game.var@1(name="falls", value=0)
lives = game.var@1(name="lives", value=3)
ground = game.ground@1(y=-30)

# the flow: a title, the game, a pause, and the two ways it ends; Enter or A starts and restarts, Esc or Start pauses
st_title = game.state@1(name="title", frozen=true)
st_play = game.state@1(name="playing")
st_pause = game.state@1(name="paused", frozen=true)
st_over = game.state@1(name="game_over", frozen=true)
st_win = game.state@1(name="win", frozen=true)
start = game.input@1(name="start", key="press", target="tag:hero", keys="Enter,Space", pad="button0", label="Start")
pause = game.input@1(name="pause", key="press", target="tag:hero", keys="Escape,KeyP", pad="button9", label="Pause")
r_start = game.rule@1(on="input:start", state="title", do="go:playing")
r_pause = game.rule@1(on="input:pause", state="playing", do="go:paused")
r_resume = game.rule@1(on="input:pause", state="paused", do="go:playing")
r_again = game.rule@1(on="input:start", state="game_over", do="restart:title")
r_again_win = game.rule@1(on="input:start", state="win", do="restart:title")

# the screens, all from game.ui nodes: Enter or A presses the focused button, the arrows move the focus, a click presses
t_menu = game.ui@1(name="t_menu", type="stack", anchor="c", align="center", gap=0.03, pad=0.05, fill="rgba(8,12,20,0.78)", border="rgba(255,255,255,0.25)", show="title")
t_name = game.ui@1(name="t_name", type="text", parent="t_menu", text="SKY RUN", letter=0.09, color="#ffd76a")
t_play = game.ui@1(name="t_play", type="button", parent="t_menu", text="Play", size="0.3,0.075", do="go:playing", tip="Run, jump, collect three coins")
t_help = game.ui@1(name="t_help", type="text", parent="t_menu", text="{key:move} to run   {key:jump} to jump (again in the air)   {key:view} changes the view", letter=0.026)
h_coins = game.ui@1(name="h_coins", type="text", anchor="tl", text="Coins {score} / 3", letter=0.04, plate="true", show="playing")
h_lives = game.ui@1(name="h_lives", type="bar", anchor="tr", size="0.2,0.03", value="lives", max=3, color="#e8505b", label="lives", show="playing")
p_menu = game.ui@1(name="p_menu", type="stack", anchor="c", align="center", gap=0.025, pad=0.04, fill="rgba(8,12,20,0.82)", show="paused")
p_name = game.ui@1(name="p_name", type="text", parent="p_menu", text="Paused", letter=0.06)
p_resume = game.ui@1(name="p_resume", type="button", parent="p_menu", text="Resume", size="0.3,0.07", do="go:playing")
p_quit = game.ui@1(name="p_quit", type="button", parent="p_menu", text="Quit to title", size="0.3,0.07", do="restart:title")
o_menu = game.ui@1(name="o_menu", type="stack", anchor="c", align="center", gap=0.025, pad=0.05, fill="rgba(30,8,12,0.85)", show="game_over")
o_name = game.ui@1(name="o_name", type="text", parent="o_menu", text="Out of lives", letter=0.07, color="#e8505b")
o_again = game.ui@1(name="o_again", type="button", parent="o_menu", text="Try again", size="0.3,0.07", do="restart:playing")
w_menu = game.ui@1(name="w_menu", type="stack", anchor="c", align="center", gap=0.025, pad=0.05, fill="rgba(8,24,14,0.85)", show="win")
w_name = game.ui@1(name="w_name", type="text", parent="w_menu", text="All three coins!", letter=0.07, color="#7ee08a")
w_falls = game.ui@1(name="w_falls", type="text", parent="w_menu", text="Falls: {falls}", letter=0.035)
w_again = game.ui@1(name="w_again", type="button", parent="w_menu", text="Play again", size="0.3,0.07", do="restart:playing")

move = game.input@1(name="move", axis="x,z", target="tag:hero", keys="KeyD,KeyA,KeyW,KeyS", pad="stick:left", label="Run")
jump = game.input@1(name="jump", key="press", target="tag:hero", keys="Space,KeyC", pad="button0", label="Jump")
ctl = game.controller@1(kind="player", mode="platformer", speed=6, accel=60, decel=70, jump=9.5, gravity=28, extra_jumps=1, relative="camera", fall_limit=-8)
cam = game.camera@1(name="side", mode="side", distance=12, height=1.5, yaw=180, deadzone=1.5, smoothing=0.15, bounds="0,25,-1,1")
# V (or Y on a pad) cycles the view: the side-scroller, then behind the hero, then through its eyes; click the view to look around with the mouse
cam_third = game.camera@1(name="behind", mode="third", distance=5, height=1.5, shoulder=0.4, smoothing=0.08)
cam_first = game.camera@1(name="eyes", mode="first", height=1.3)
view = game.input@1(name="view", key="press", target="tag:hero", keys="KeyV", pad="button3", label="View")
r_view = game.rule@1(on="input:view", do="effect:camera:next:0.4")
take = game.rule@1(on="touch", a="tag:hero", b="tag:coin", do="add:score:1;destroy:b")
fell = game.rule@1(on="fell", do="add:falls:1;add:lives:-1")
won = game.goal@1(outcome="win", when="score>=3")
lost = game.goal@1(outcome="lose", when="lives<=0")

all = game.collect@1(t_menu, t_name, t_play, t_help, h_coins, h_lives, p_menu, p_name, p_resume, p_quit, o_menu, o_name, o_again, w_menu, w_name, w_falls, w_again, k_hero, k_base, k_plat, k_coin, s_hero, s_p0, s_p1, s_p2, s_p3, s_p4, c1, c2, c3, score, falls, lives, ground, st_title, st_play, st_pause, st_over, st_win, start, pause, r_start, r_pause, r_resume, r_again, r_again_win, move, jump, ctl, cam, cam_third, cam_first, view, r_view, take, fell, won, lost, name="platformer")
output rules = all
