# shared helpers: piecewise-linear lookup, a ring mirrored from its right half, and a NACA thickness ordinate
let jsh = "const pl = (k, z) => { if (z <= k[0][0]) return k[0][1]; for (let i = 1; i < k.length; i++) { if (z <= k[i][0]) { const t = (z - k[i - 1][0]) / (k[i][0] - k[i - 1][0]); return k[i - 1][1] + (k[i][1] - k[i - 1][1]) * t; } } return k[k.length - 1][1]; }; const ring = (rh) => { const r = rh.slice(); for (let i = rh.length - 2; i >= 1; i--) r.push([-rh[i][0], rh[i][1]]); return r; }; const naca = (xc, t) => { const s = Math.sqrt(Math.max(xc, 0)); return 5 * t * (0.2969 * s - 0.126 * xc - 0.3516 * xc * xc + 0.2843 * xc * xc * xc - 0.1015 * xc * xc * xc * xc); };"

# ---- FUSELAGE: one loft of rounded-box sections from the tail cone to the firewall. Widths, heights and the rising
