# Sources: Wikipedia and the US Navy fact file (sizes); the sources disagree on what height the masts are measured from, so heights here are above the waterline, scaled from
# the 198 / 220 / 172.5 ft figures and anchored to 172 ft from the spar deck to the main truck (= 57.3 m above the water). Yard lengths and mast positions are ESTIMATED from model plans.

# are tables), subdivided; the deck, bulwarks and gun ports are cut out of it and each opening is filled with its own material. engine/tools/audit-ship.ts measures the
# evaluated mesh against the real dimensions and says which value to change.
# Metres, +y up, the ship faces +z (bow), waterline at y = 0, midships near z = 0.

# shared by the hull, deck and ports: piecewise-linear lookup of the body plan, and the half-width of the hull at a height
let jsh = "const pl = (k, z) => { if (z <= k[0][0]) return k[0][1]; for (let i = 1; i < k.length; i++) { if (z <= k[i][0]) { const t = (z - k[i - 1][0]) / (k[i][0] - k[i - 1][0]); return k[i - 1][1] + (k[i][1] - k[i - 1][1]) * t; } } return k[k.length - 1][1]; }; const BEAM = ${beam}, DA = ${draft_aft}, DF = ${draft_fwd}, FB = ${freeboard}; const bK = [[-29, 3.9], [-27, 4.7], [-24, 5.7], [-16, 6.45], [-8, 6.63], [0, 6.63], [8, 6.5], [16, 5.9], [22, 4.6], [26, 2.4], [28, 1.3], [30, 0.7], [33, 0.2]]; const yK = [[-29, 3.0], [-26.5, 0], [-24, -DA * 0.57], [-20, -DA * 0.99], [-10, -DA], [19, -DF], [23, -DF * 0.7], [26.8, 0], [29, 2.5], [31, 4.5], [33, 6.5]]; const dK = [[-29, FB + 1.4], [-9.2, FB + 1.4], [-8, FB], [21, FB], [22.2, FB + 1.4], [33, FB + 1.4]]; const half = (z) => pl(bK, z) * BEAM / 13.26; const railY = (z) => pl(dK, z) + 1.15; const fS = [[0, 1.0], [0.22, 1.012], [0.45, 1.0], [0.70, 0.965], [1, 0.935]]; const y0Of = (z) => Math.min(Math.max(0, pl(yK, z) + (pl(yK, z) > 0 ? 1.0 : 0)), railY(z) - 0.3); const sideX = (z, y) => { const y0 = y0Of(z), r = railY(z); return half(z) * pl(fS, Math.min(1, Math.max(0, (y - y0) / (r - y0)))); }; const ring = (rh) => { const r = rh.slice(); for (let i = rh.length - 2; i >= 1; i--) r.push([-rh[i][0], rh[i][1]]); return r; };"
# the hull rings: the keel line up to the waterline in eight points (U-shaped section), then tumblehome up to the rail, then across the top
let hull_code = "${jsh} const m = new Mesh(); const S = [0, 0.22, 0.45, 0.66, 0.82, 0.93, 0.985, 1.0], Hh = [0, 0.035, 0.10, 0.22, 0.42, 0.65, 0.87, 1.0], U = [0.22, 0.45, 0.70, 1.0]; const fn = (z) => { const b = half(z), yk = pl(yK, z), r = railY(z), y0 = y0Of(z); const out = []; for (let i = 0; i < S.length; i++) out.push([b * S[i], yk + (y0 - yk) * Hh[i]]); for (let i = 0; i < U.length; i++) out.push([b * pl(fS, U[i]), y0 + (r - y0) * U[i]]); const e = b * 0.935, d = Math.min(0.2, e * 0.4); out.push([e - d, r]); out.push([e * 0.5, r]); out.push([0, r]); return out; }; const zs = []; const N = 64; for (let i = 0; i <= N; i++) zs.push(-29 + 62 * i / N); const rows = zs.map((z) => ring(fn(z)).map((q) => m.vertex(q[0], q[1], z))); const n = rows[0].length; for (let k = 0; k < rows.length - 1; k++) for (let j = 0; j < n; j++) { const j2 = (j + 1) % n; m.face(rows[k][j], rows[k][j2], rows[k + 1][j2], rows[k + 1][j]); } const cf = m.vertex(0, (pl(yK, 33) + railY(33)) / 2, 33), cb = m.vertex(0, (pl(yK, -29) + railY(-29)) / 2, -29); for (let j = 0; j < n; j++) { const j2 = (j + 1) % n; m.face(cf, rows[rows.length - 1][j], rows[rows.length - 1][j2]); m.face(cb, rows[0][j2], rows[0][j]); } m"
# the open deck: a loft inset by the bulwark thickness (0.3 m), from 0.08 m below the deck up past the rail (cut), and the same 0.08 m plate (the deck planking); steps follow the deck table
let deck_body = "${jsh} const m = new Mesh(); const zs = []; const N = 90; for (let i = 0; i <= N; i++) zs.push(-28.7 + 60.1 * i / N); for (const e of [-9.2, -8.6, -8, 21, 21.6, 22.2]) zs.push(e); zs.sort((a, b) => a - b); for (let i = zs.length - 1; i > 0; i--) if (zs[i] - zs[i - 1] < 0.08) zs.splice(i, 1); const fn = (z) => { const bi = Math.max(half(z) * 0.935 - 0.3, 0.05), dy = pl(dK, z); return [[0, dy - 0.08], [bi * 0.5, dy - 0.08], [bi, dy - 0.08], [bi, TOP(z, dy)], [bi * 0.5, TOP(z, dy)], [0, TOP(z, dy)]]; }; const rows = zs.map((z) => ring(fn(z)).map((q) => m.vertex(q[0], q[1], z))); const n = rows[0].length; for (let k = 0; k < rows.length - 1; k++) for (let j = 0; j < n; j++) { const j2 = (j + 1) % n; m.face(rows[k][j], rows[k][j2], rows[k + 1][j2], rows[k + 1][j]); } const cf = m.vertex(0, (fn(zs[zs.length - 1])[0][1] + fn(zs[zs.length - 1])[3][1]) / 2, zs[zs.length - 1]), cb = m.vertex(0, (fn(zs[0])[0][1] + fn(zs[0])[3][1]) / 2, zs[0]); for (let j = 0; j < n; j++) { const j2 = (j + 1) % n; m.face(cf, rows[rows.length - 1][j], rows[rows.length - 1][j2]); m.face(cb, rows[0][j2], rows[0][j]); } m"

# a rope or spar: an n-sided prism between two points (n = 4 for a rope, 10 for a spar), closed and outward-facing
let jsrope = "const rope = (m, a, b, r, n) => { const d = [b[0] - a[0], b[1] - a[1], b[2] - a[2]]; const L = Math.hypot(d[0], d[1], d[2]); const t = [d[0] / L, d[1] / L, d[2] / L]; const up = Math.abs(t[1]) > 0.9 ? [1, 0, 0] : [0, 1, 0]; let n1 = [t[1] * up[2] - t[2] * up[1], t[2] * up[0] - t[0] * up[2], t[0] * up[1] - t[1] * up[0]]; const l1 = Math.hypot(n1[0], n1[1], n1[2]); n1 = n1.map((v) => v / l1); const n2 = [t[1] * n1[2] - t[2] * n1[1], t[2] * n1[0] - t[0] * n1[2], t[0] * n1[1] - t[1] * n1[0]]; const A = [], B = []; for (let k = 0; k < n; k++) { const th = 2 * Math.PI * k / n, c = Math.cos(th), s = Math.sin(th); const o = [r * (c * n1[0] + s * n2[0]), r * (c * n1[1] + s * n2[1]), r * (c * n1[2] + s * n2[2])]; A.push(m.vertex(a[0] + o[0], a[1] + o[1], a[2] + o[2])); B.push(m.vertex(b[0] + o[0], b[1] + o[1], b[2] + o[2])); } for (let k = 0; k < n; k++) { const k2 = (k + 1) % n; m.face(A[k], A[k2], B[k2]); m.face(A[k], B[k2], B[k]); } m.face(...A.slice().reverse()); m.face(...B); };"

# a mast: four tapered segments (lower, top, topgallant, royal pole) revolved, a top platform, raked aft; r0 is the radius at the deck
def mast(tag, mz, mh, r0, rake)
  mp{tag}_a = profile.polygon@1(points=[[0, 4.0], [{r0}, 4.0], [{r0 * 0.72}, {mh * 0.5}], [0, {mh * 0.5}]])
  mp{tag}_b = profile.polygon@1(points=[[0, {mh * 0.42}], [{r0 * 0.58}, {mh * 0.42}], [{r0 * 0.42}, {mh * 0.78}], [0, {mh * 0.78}]])
  mp{tag}_c = profile.polygon@1(points=[[0, {mh * 0.72}], [{r0 * 0.33}, {mh * 0.72}], [{r0 * 0.23}, {mh * 0.93}], [0, {mh * 0.93}]])
  mp{tag}_d = profile.polygon@1(points=[[0, {mh * 0.89}], [{r0 * 0.19}, {mh * 0.89}], [{r0 * 0.09}, {mh}], [0, {mh}]])
  mr{tag}_a = solid.revolve@1(mp{tag}_a, angle=360deg, segments=16)
  mr{tag}_b = solid.revolve@1(mp{tag}_b, angle=360deg, segments=14)
  mr{tag}_c = solid.revolve@1(mp{tag}_c, angle=360deg, segments=12)
  mr{tag}_d = solid.revolve@1(mp{tag}_d, angle=360deg, segments=10)
  mt{tag}_p = solid.box@1(size=[{mh * 0.062}, 0.2, {mh * 0.05}])
  mt{tag}_q = xform.move@1(mt{tag}_p, by=[0, {mh * 0.5}, 0.2])
  mu{tag} = solid.union@1(mr{tag}_a, mr{tag}_b, mr{tag}_c, mr{tag}_d, mt{tag}_q)
  mk{tag} = xform.rotate@1(mu{tag}, angles=[{0 - rake}, 0, 0])
  mm{tag} = xform.move@1(mk{tag}, by=[0, 0, {mz}])
  mc{tag} = solid.material@1(mm{tag}, color=[0.5, 0.36, 0.2], roughness=0.8)
  mq{tag} = anim.parent@1(mc{tag}, parent="hull")
  output mast_{tag} = mq{tag}
end

let square_src = "sim sail\nparam wind = vec3(1.4, 0.0, 6.0)\nparam gust = 0.6\nparam damping = 0.98\nparam top = 0.0\nparam bot = 0.0\nparam hw = 1.0\ninput mesh sail\ninput bone hull\nstate pos: vec3[sail]\nstate prev: vec3[sail]\nkernel init each i in pos {\n  let w = bone_point(hull, rest(sail, i))\n  pos[i] = w\n  prev[i] = w\n}\nkernel predict each i in pos {\n  let r = rest(sail, i)\n  let pin = r.y > top - 0.05 || (r.y < bot + 0.05 && abs(r.x) > hw - 0.5)\n  let w = bone_point(hull, r)\n  if pin || distance(pos[i], w) > 8.0 {\n    pos[i] = w\n    prev[i] = w\n  } else {\n    let v = (pos[i] - prev[i]) * damping\n    v = v * min(1.0, 0.12 / max(length(v), 0.0001))\n    prev[i] = pos[i]\n    let puff = wind * (1.0 + gust * sin(t * 1.7 + r.y * 0.6 + r.x * 0.5))\n    pos[i] += v + (vec3(0.0, -9.8, 0.0) + puff * 2.4) * dt * dt\n  }\n}\nkernel solve each i in pos {\n  let r = rest(sail, i)\n  let pin = r.y > top - 0.05 || (r.y < bot + 0.05 && abs(r.x) > hw - 0.5)\n  let p = pos[i]\n  if !pin {\n    let corr = vec3(0.0)\n    let count = 0.0\n    for j, len in adjacent(sail, i) {\n      let rj = rest(sail, j)\n      let pj = rj.y > top - 0.05 || (rj.y < bot + 0.05 && abs(rj.x) > hw - 0.5)\n      corr += pull(p, pos[j], len) * select(pj, 1.0, 0.5)\n      count += 1.0\n    }\n    if count > 0.0 {\n      p += corr * (1.3 / count)\n    }\n  }\n  pos[i] = p\n}\nschedule predict; repeat 6 { solve }\noutput mesh pos\n"

buoy = sim.program@1(size=0.03, rate=60, source="sim buoyancy\nparam spring = 7.0\nparam damp = 3.2\nparam gravity = 9.8\ninput bone hull\nstate probe: vec3[1]\nstate push: vec3[1]\nstate last: vec3[1]\nkernel feel each i in probe {\n  let c = bone_point(hull, vec3(0.0, 0.0, 0.0))\n  let h = 2.0 + 0.9 * sin(0.21 * c.x + 0.13 * c.z - 0.9 * t) + 0.5 * sin(-0.11 * c.x + 0.27 * c.z - 1.3 * t + 1.7) + 0.22 * sin(0.43 * c.x + 0.37 * c.z - 2.1 * t + 0.6)\n  let vy = clamp((c.y - last[i].y) / dt, -12.0, 12.0)\n  let err = clamp(h - c.y, -2.5, 2.5)\n  probe[i] = c\n  last[i] = c\n  push[i] = vec3(0.0, clamp((gravity + spring * err - damp * vy) * dt, -0.2, 0.3), 0.0)\n}\nschedule feel\noutput particles probe\noutput push hull push\n")
buoy_m = solid.material@1(buoy, color=[1, 1, 0.2], roughness=0.5)
output hull_probe = buoy_m

ship1 = game.spawn@1(kind="frigate", at=[0, 2, 0], heading=0)
rules = game.collect@1(kind_ship, ship1, name="frigate_game")
output rules = rules
