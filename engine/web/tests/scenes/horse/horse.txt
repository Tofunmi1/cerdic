# HORSE: a bay stallion, 2.9 m nose to tail, in the default studio, walking in place. Everything is built in our own language: the body is one field (barrel, chest, rump,
# neck, head, shoulder, thigh and forearm muscles, upper legs) smooth-united and mirrored, the lower legs and hooves are separate parts, the mane and tail are
# thousands of hair wisps, the skeleton is 20 bones, and the walk is the four-beat cycle of a real horse (left hind, left fore, right hind, right fore).

let bones = "pelvis; spine; neck1; neck2; head; lf_up; lf_can; lf_pas; rf_up; rf_can; rf_pas; lh_up; lh_can; lh_pas; rh_up; rh_can; rh_pas; tail1; tail2; tail3"

# A bone that is only a joint.
def bn(tag, hx, hy, hz, tx, ty, tz, par)
  bn{tag}_a = solid.box@1(size=[0.01, 0.01, 0.01])
  bn{tag}_b = xform.move@1(bn{tag}_a, by=[{hx}, {hy}, {hz}])
  bn{tag}_c = rig.bone@1(bn{tag}_b, head=[{hx}, {hy}, {hz}], tail=[{tx}, {ty}, {tz}])
  bn{tag}_d = anim.parent@1(bn{tag}_c, parent="${par}")
  output {tag} = bn{tag}_d
end
# A bone that swings about the side axis: amp degrees around base, period seconds, phase of the cycle.
def bw(tag, hx, hy, hz, tx, ty, tz, par, amp, base, per, ph)
  bw{tag}_a = solid.box@1(size=[0.01, 0.01, 0.01])
  bw{tag}_b = xform.move@1(bw{tag}_a, by=[{hx}, {hy}, {hz}])
  bw{tag}_c = rig.bone@1(bw{tag}_b, head=[{hx}, {hy}, {hz}], tail=[{tx}, {ty}, {tz}])
  bw{tag}_d = anim.parent@1(bw{tag}_c, parent="${par}")
  bw{tag}_e = anim.wave@1(bw{tag}_d, property="rotation", amplitude=[0, 0, {amp}], base=[0, 0, {base}], period={per}, phase={ph}, pivot_at="head")
  output {tag} = bw{tag}_e
end

# HAIR: each wisp is a tapered ribbon that leaves the skin, arcs and falls, turned a random way about the vertical so the strands cross. Mane along the crest of the neck,
# a forelock over the brow, a full tail from the dock.
let jhair = "const m = new Mesh(); let sd = 7; function rnd(){ sd = (sd * 1664525 + 1013904223) % 4294967296; return sd / 4294967296; } function wisp(P, d, L, droop, w, rows){ const a0 = rnd() * 3.14159, sx = Math.cos(a0), sz = Math.sin(a0); let prev = null; for (let i = 0; i <= rows; i++){ const t = i / rows, x = P[0] + d[0] * L * t + 0.03 * L * Math.sin(t * 5 + a0), y = P[1] + d[1] * L * t - droop * L * t * t, z = P[2] + d[2] * L * t, ww = w * (1 - 0.8 * t); const a = m.vertex(x - sx * ww, y, z - sz * ww), b = m.vertex(x + sx * ww, y, z + sz * ww); if (prev){ m.face(prev[0], prev[1], b, a); m.face(prev[0], a, b, prev[1]); } prev = [a, b]; } }"
mane_s = poly.script@1(code="${jhair} for (let k = 0; k < 1200; k++){ const u = rnd(), v = 1 - u; const P = [v * v * 0.6 + 2 * u * v * 0.9 + u * u * 1.2, v * v * 1.7 + 2 * u * v * 2.02 + u * u * 2.32 - 0.03, (rnd() - 0.5) * 0.05]; const side = rnd() < 0.85 ? 1 : -1; wisp(P, [0.12 + 0.15 * rnd(), 0.4 + 0.2 * rnd(), side * (0.7 + 0.4 * rnd())], 0.2 + 0.22 * rnd() + 0.12 * v, 1.0 + 0.3 * rnd(), 0.012, 6); } for (let k = 0; k < 140; k++){ const P = [1.24 + 0.02 * rnd(), 2.34, (rnd() - 0.5) * 0.12]; wisp(P, [0.55, -0.25 + 0.3 * rnd(), (rnd() - 0.5) * 0.4], 0.16 + 0.14 * rnd(), 0.5, 0.01, 5); } m", budget=20000)
mane_m = solid.material@1(mane_s, color=[0.03, 0.022, 0.018], roughness=0.5)
mane_f = solid.finish@1(mane_m, sheen=0.6, clearcoat=0.15, clearcoat_roughness=0.3)
mane_k = rig.skin@1(mane_f, bones="neck1; neck2; head; spine", falloff=3, influences=2)
output horse_mane = mane_k
