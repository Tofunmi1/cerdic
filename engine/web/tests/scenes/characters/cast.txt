# ---- species bodies: smooth fields, chunky proportions. name, x, H (height), sw (shoulder half width), hw (hip half width), hr (head radius), belly, hunch, skin r g b ----
def person(name, x, H, sw, hw, hr, belly, hunch, r, g, b)
  n_{name}_torso = field.ellipsoid@1(radii=[{sw * 0.98}, {H * 0.2}, {H * 0.115 + belly}])
  n_{name}_torso_m = field.move@1(n_{name}_torso, by=[0, {H * 0.63}, {hunch * 0.4}])
  n_{name}_pelvis = field.ellipsoid@1(radii=[{hw * 1.55}, {H * 0.08}, {H * 0.1}])
  n_{name}_pelvis_m = field.move@1(n_{name}_pelvis, by=[0, {H * 0.5}, 0])
  n_{name}_neck = field.tube@1(points=[[0, {H * 0.76}, {hunch * 0.5}], [0, {H * 0.84}, {hunch}]], radius={H * 0.05}, end_radius={H * 0.045})

  n_{name}_head_m = field.move@1(n_{name}_head, by=[0, {H * 0.885}, {hunch}])
  n_{name}_jaw = field.ellipsoid@1(radii=[{hr * 0.7}, {hr * 0.55}, {hr * 0.8}])
  n_{name}_jaw_m = field.move@1(n_{name}_jaw, by=[0, {H * 0.885 - hr * 0.55}, {hunch + hr * 0.2}])
  n_{name}_nose = field.ellipsoid@1(radii=[{hr * 0.16}, {hr * 0.2}, {hr * 0.25}])
  n_{name}_nose_m = field.move@1(n_{name}_nose, by=[0, {H * 0.885 - hr * 0.1}, {hunch + hr * 0.95}])
  n_{name}_leg_l = field.tube@1(points=[[{hw}, {H * 0.5}, 0], [{hw * 1.05}, {H * 0.27}, {H * 0.01}], [{hw * 1.05}, {H * 0.05}, 0]], radius={H * 0.085}, end_radius={H * 0.062})
  n_{name}_leg_r = field.tube@1(points=[[{0 - hw}, {H * 0.5}, 0], [{0 - hw * 1.05}, {H * 0.27}, {H * 0.01}], [{0 - hw * 1.05}, {H * 0.05}, 0]], radius={H * 0.085}, end_radius={H * 0.062})
  n_{name}_foot = field.ellipsoid@1(radii=[{H * 0.06}, {H * 0.035}, {H * 0.11}])
  n_{name}_foot_l = field.move@1(n_{name}_foot, by=[{hw * 1.05}, {H * 0.035}, {H * 0.05}])
  n_{name}_foot_r = field.move@1(n_{name}_foot, by=[{0 - hw * 1.05}, {H * 0.035}, {H * 0.05}])
  n_{name}_arm_l = field.tube@1(points=[[{sw}, {H * 0.74}, 0], [{sw + H * 0.04}, {H * 0.59}, {H * 0.03}], [{sw + H * 0.06}, {H * 0.46}, {H * 0.1}]], radius={H * 0.055}, end_radius={H * 0.045})
  n_{name}_arm_r = field.tube@1(points=[[{0 - sw}, {H * 0.74}, 0], [{0 - sw - H * 0.04}, {H * 0.59}, {H * 0.03}], [{0 - sw - H * 0.06}, {H * 0.46}, {H * 0.1}]], radius={H * 0.055}, end_radius={H * 0.045})
  n_{name}_hand = field.sphere@1(radius={H * 0.05})
  n_{name}_hand_l = field.move@1(n_{name}_hand, by=[{sw + H * 0.06}, {H * 0.44}, {H * 0.1}])
  n_{name}_hand_r = field.move@1(n_{name}_hand, by=[{0 - sw - H * 0.06}, {H * 0.44}, {H * 0.1}])
  n_{name}_s1 = field.smooth_union@1(n_{name}_torso_m, n_{name}_pelvis_m, radius={H * 0.06})
  n_{name}_s2 = field.smooth_union@1(n_{name}_s1, n_{name}_neck, radius={H * 0.04})
  n_{name}_s3 = field.smooth_union@1(n_{name}_s2, n_{name}_head_m, radius={H * 0.03})
  n_{name}_s4 = field.smooth_union@1(n_{name}_s3, n_{name}_jaw_m, radius={H * 0.03})
  n_{name}_s5 = field.smooth_union@1(n_{name}_s4, n_{name}_nose_m, radius={H * 0.015})
  n_{name}_s6 = field.smooth_union@1(n_{name}_s5, n_{name}_leg_l, radius={H * 0.04})
  n_{name}_s7 = field.smooth_union@1(n_{name}_s6, n_{name}_leg_r, radius={H * 0.04})
  n_{name}_s8 = field.smooth_union@1(n_{name}_s7, n_{name}_foot_l, radius={H * 0.02})
  n_{name}_s9 = field.smooth_union@1(n_{name}_s8, n_{name}_foot_r, radius={H * 0.02})

  n_{name}_skin = solid.material@1(n_{name}_solid, color=[{r}, {g}, {b}], roughness=0.65)
  n_{name}_body = xform.move@1(n_{name}_skin, by=[{x}, 0, 0])
  output {name}_body = n_{name}_body
end

# ---- the kits as macros: one call per script, material and placement. name, x, H, S (shoulder half width), W (hip half width), then colours ----
def kit(name, kind, part, x, H, S, W, r, g, b, metal, e)
  n_{name}_{part} = poly.script@1(code="const m = new Mesh(); ${jall} ${kind} kit(${H},${S},${W}); m", budget=900)
  n_{name}_{part}_m = solid.material@1(n_{name}_{part}, color=[{r}, {g}, {b}], roughness=0.5, metallic={metal}, emissive=[{r * e}, {g * e}, {b * e}])
  n_{name}_{part}_p = xform.move@1(n_{name}_{part}_m, by=[{x}, 0, 0])
  output {name}_{part} = n_{name}_{part}_p
end
