# an ellipsoid at a place (the field is f_<tag>); a tapered tube between two points (also f_<tag>)
def fe(tag, cx, cy, cz, rx, ry, rz)
  e_{tag}_f = field.ellipsoid@1(radii=[{rx}, {ry}, {rz}])
  f_{tag} = field.move@1(e_{tag}_f, by=[{cx}, {cy}, {cz}])
end
def ft(tag, x0, y0, z0, x1, y1, z1, r0, r1)
  f_{tag} = field.tube@1(points=[[{x0}, {y0}, {z0}], [{x1}, {y1}, {z1}]], radius={r0}, end_radius={r1})
end
