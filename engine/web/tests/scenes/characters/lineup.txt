# THE RANGE OF GAME CHARACTERS, one scene, three kinds: an ORGANIC stylized adventurer (5 heads tall, skin body with separate clothing shells: tunic, trousers, boots, gloves, belt, hair, sword), an ANIMAL (a fox: spine, four legs, tail,
# ears and snout from one skeleton), and a MECHANICAL robot (rigid parts on joints: housings, pistons, a glowing core). The two living ones come from poly.skin (Blender's Skin modifier: a skeleton of named joints with radii becomes
# one smooth quad surface); clothes are the same kind of skin with slightly larger radii over part of the skeleton, so they fit by construction. Toon shading for the stylized pair, PBR metal for the machine.
# Metres, +y up, facing +z. Adventurer at x = -1.3, fox at x = 0.5, robot at x = 2.0.
# kind: world
sky = scene.sky@1(sun_elevation=38, sun_azimuth=-35, sun_intensity=3.4, turbidity=3, ground_color=[0.32, 0.3, 0.28], clouds=false, haze=0.08)

output sky = sky

# a skin shell: a skeleton skinned into one surface, smoothed, moved to its place, toon shaded. `tag` names the output; px is where the character stands.
def shell(tag, px, js, bs, cr, cg, cb)
  sh{tag}_s = poly.skin@1(joints="${js}", bones="${bs}", symmetry="x")
  sh{tag}_d = poly.subdivide@1(sh{tag}_s, levels=2)
  sh{tag}_p = xform.move@1(sh{tag}_d, by=[{px}, 0, 0])
  sh{tag}_m = solid.material@1(sh{tag}_p, color=[{cr}, {cg}, {cb}], roughness=0.65)
  sh{tag}_t = solid.toon@1(sh{tag}_m, bands=3, feather=0.12, outline=1.3, shade_color=[{cr * 0.45}, {cg * 0.45}, {cb * 0.55}], rim=0.25, spec=0.35, outline_color=[0.05, 0.04, 0.07])
  output {tag} = sh{tag}_t
end
