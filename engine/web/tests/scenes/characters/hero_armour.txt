# THE HEROINE'S HAIR, ARMOUR, SWORD AND CAPE: rigid parts on the cut-body segments (assemble after hero_cut.txt). The shape to dress is the node named x<tag>.
# xform.mirror keeps the original AND adds the mirror image, so a right-hand plate is the mirror cut to x < 0.
rh_b = solid.box@1(size=[2, 4, 4])
rhalf = xform.move@1(rh_b, by=[-1, 1, 0])
def mir(tag)
  mr_{tag} = xform.mirror@1(x{tag}_l, axis="x")
  x{tag}_r = solid.intersect@1(mr_{tag}, rhalf)
end
def steel(tag, parent)
  as_{tag}_m = solid.material@1(x{tag}, color=[0.6, 0.64, 0.72], metallic=0.85, roughness=0.32)
  as_{tag}_t = solid.toon@1(as_{tag}_m, bands=3, feather=0.1, outline=1.4, shade_color=[0.34, 0.4, 0.56], rim=0.3, spec=0.5, outline_color=[0.05, 0.06, 0.1])
  as_{tag}_p = anim.parent@1(as_{tag}_t, parent={parent})
  output a_{tag} = as_{tag}_p
end
def gold(tag, parent)
  ag_{tag}_m = solid.material@1(x{tag}, color=[1.0, 0.78, 0.3], metallic=0.9, roughness=0.28)
  ag_{tag}_t = solid.toon@1(ag_{tag}_m, bands=3, feather=0.1, outline=1.0, shade_color=[0.62, 0.42, 0.28], rim=0.3, spec=0.6, outline_color=[0.25, 0.14, 0.04])
  ag_{tag}_p = anim.parent@1(ag_{tag}_t, parent={parent})
  output g_{tag} = ag_{tag}_p
end
