# A HEROINE BUILT ONLY FROM THE LANGUAGE, WITH NO SKINNING: rigid parts on a joint hierarchy, so the game can place and move her anywhere.
# Limbs are convex hulls between two spheres (a tapered segment between any two joints, the spheres hide the joint seams); torso, pelvis and head are smooth unions of ellipsoids meshed from fields.
# Metres, 1.7 m tall, facing +z, y up. Joint positions follow the human pack's proportions.

# skin look shared by every part (a macro so a part is three lines)
def dress(tag, par)
  sk_{tag}_m = solid.material@1(b{tag}, color=[0.86, 0.64, 0.52], roughness=0.5)
  sk_{tag}_t = solid.toon@1(sk_{tag}_m, bands=3, feather=0.2, outline=1.0, shade_color=[0.93, 0.7, 0.66], rim=0.25, outline_color=[0.2, 0.08, 0.06])
  sk_{tag}_p = anim.parent@1(sk_{tag}_t, parent={par})
  output {tag} = sk_{tag}_p
end
# a limb segment: hull of a sphere at joint A and a sphere at joint B (radii ra, rb), plus a muscle bulge sphere at a fraction along it
def limb(tag, par, ax, ay, az, bx, by, bz, ra, rb, rm)
  ls_{tag}_a = solid.sphere@1(radius={ra}, segments=20)
  ls_{tag}_am = xform.move@1(ls_{tag}_a, by=[{ax}, {ay}, {az}])
  ls_{tag}_b = solid.sphere@1(radius={rb}, segments=20)
  ls_{tag}_bm = xform.move@1(ls_{tag}_b, by=[{bx}, {by}, {bz}])
  ls_{tag}_c = solid.sphere@1(radius={rm}, segments=20)
  ls_{tag}_cm = xform.move@1(ls_{tag}_c, by=[{(ax + bx) / 2}, {(ay + by) / 2}, {(az + bz) / 2}])
  ls_{tag}_w = curve.polyline@1(points=[[{ax}, {ay}, {az}], [{bx}, {by}, {bz}]])
  ls_{tag}_q = profile.circle@1(radius={ra}, segments=20)
  ls_{tag}_t = solid.sweep@1(ls_{tag}_w, ls_{tag}_q, scale_end={rb / ra})
  ls_{tag}_u0 = solid.union@1(ls_{tag}_t, ls_{tag}_am)
  ls_{tag}_u1 = solid.union@1(ls_{tag}_u0, ls_{tag}_bm)
  b{tag} = solid.union@1(ls_{tag}_u1, ls_{tag}_cm)
  sk_{tag}_m = solid.material@1(b{tag}, color=[0.86, 0.64, 0.52], roughness=0.5)
  sk_{tag}_t = solid.toon@1(sk_{tag}_m, bands=3, feather=0.2, outline=1.0, shade_color=[0.93, 0.7, 0.66], rim=0.25, outline_color=[0.2, 0.08, 0.06])
  sk_{tag}_p = anim.parent@1(sk_{tag}_t, parent={par})
  output {tag} = sk_{tag}_p
end
