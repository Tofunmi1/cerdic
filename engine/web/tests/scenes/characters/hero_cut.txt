# THE HEROINE ON THE REAL BODY, CUT INTO RIGID SEGMENTS (no skinning, so the game places and moves her anywhere without distortion).
# The human pack's body is sliced by boxes at the joints; every segment is a part on a joint hierarchy with its own clips (rotation about the joint),
# spheres cover the joints, and armour, sword, cape and eyes are rigid parts on the segments. Dark suit and boots are colour regions of the segments.
let still = "0:0,0,0; 1:0,0,0"
let skin = "0.86, 0.64, 0.52"
hbody = body.human@1(gender=0.85, age=0.667, height=0.5, weight=0.42, muscle=0.62, caucasian=0.5, asian=0.2, african=0.3)
srcb = poly.to_solid@1(hbody)

# one segment: the body clipped by a box, toon-coloured; its node is p0<tag>
def seg(tag, x0, y0, z0, x1, y1, z1, cr, cg, cb)
  sg_{tag}_b = solid.box@1(size=[{x1 - x0}, {y1 - y0}, {z1 - z0}])
  sg_{tag}_c = xform.move@1(sg_{tag}_b, by=[{(x0 + x1) / 2}, {(y0 + y1) / 2}, {(z0 + z1) / 2}])
  sg_{tag}_i = solid.intersect@1(srcb, sg_{tag}_c)
  sg_{tag}_m = solid.material@1(sg_{tag}_i, color=[{cr}, {cg}, {cb}], roughness=0.55)
  p0{tag} = solid.toon@1(sg_{tag}_m, bands=3, feather=0.2, outline=1.0, shade_color=[0.7, 0.62, 0.7], rim=0.25, outline_color=[0.12, 0.05, 0.05])
end
# the first five clips of a segment, as rotation about the joint (px, py, pz); the result is pa<tag>
def bone(tag, hx, hy, hz, tx, ty, tz)
  pq{tag} = rig.bone@1(p0{tag}, head=[{hx}, {hy}, {hz}], tail=[{tx}, {ty}, {tz}])
end
def trkA(tag, px, py, pz, idle, run, slash, slash2, heavy)
  ta_{tag}_0 = anim.track@1(pq{tag}, property="rotation", keys={idle}, interpolation="smooth", clip="idle", pivot=[{px}, {py}, {pz}])
  ta_{tag}_1 = anim.track@1(ta_{tag}_0, property="rotation", keys={run}, interpolation="smooth", clip="run", pivot=[{px}, {py}, {pz}])
  ta_{tag}_2 = anim.track@1(ta_{tag}_1, property="rotation", keys={slash}, interpolation="smooth", loop=false, clip="slash", pivot=[{px}, {py}, {pz}])
  ta_{tag}_3 = anim.track@1(ta_{tag}_2, property="rotation", keys={slash2}, interpolation="smooth", loop=false, clip="slash2", pivot=[{px}, {py}, {pz}])
  pa{tag} = anim.track@1(ta_{tag}_3, property="rotation", keys={heavy}, interpolation="smooth", loop=false, clip="heavy", pivot=[{px}, {py}, {pz}])
end
# the other five clips; the result is pb<tag>
def trkB(tag, px, py, pz, roll, hit, block, cast, death)
  tb_{tag}_0 = anim.track@1(pa{tag}, property="rotation", keys={roll}, interpolation="smooth", loop=false, clip="roll", pivot=[{px}, {py}, {pz}])
  tb_{tag}_1 = anim.track@1(tb_{tag}_0, property="rotation", keys={hit}, interpolation="smooth", loop=false, clip="hit", pivot=[{px}, {py}, {pz}])
  tb_{tag}_2 = anim.track@1(tb_{tag}_1, property="rotation", keys={block}, interpolation="smooth", loop=false, clip="block", pivot=[{px}, {py}, {pz}])
  tb_{tag}_3 = anim.track@1(tb_{tag}_2, property="rotation", keys={cast}, interpolation="smooth", loop=false, clip="cast", pivot=[{px}, {py}, {pz}])
  pb{tag} = anim.track@1(tb_{tag}_3, property="rotation", keys={death}, interpolation="smooth", loop=false, clip="death", pivot=[{px}, {py}, {pz}])
end
# hang a segment with clips from its parent, and name it; hang0 for a segment without clips
def hang(tag, par)
  hg_{tag} = anim.parent@1(pb{tag}, parent={par})
  output {tag} = hg_{tag}
end
def hang0(tag, par)
  hz_{tag} = anim.parent@1(p0{tag}, parent={par})
  output {tag} = hz_{tag}
end
# a skin sphere over a joint, riding a segment
def joint(tag, par, x, y, z, r, cr, cg, cb)
  jt_{tag}_s = solid.sphere@1(radius={r}, segments=20)
  jt_{tag}_a = xform.move@1(jt_{tag}_s, by=[{x}, {y}, {z}])
  jt_{tag}_m = solid.material@1(jt_{tag}_a, color=[{cr}, {cg}, {cb}], roughness=0.55)
  jt_{tag}_t = solid.toon@1(jt_{tag}_m, bands=3, feather=0.2, outline=1.0, shade_color=[0.7, 0.62, 0.7], rim=0.25, outline_color=[0.12, 0.05, 0.05])
  jt_{tag}_p = anim.parent@1(jt_{tag}_t, parent={par})
  output j_{tag} = jt_{tag}_p
end
