def tree(tag, tx, tz)
  t{tag}_k0 = field.tube@1(points=[[0, 0, 0], [0.08, 2.4, 0.05], [0, 4.0, 0]], radius=0.26, end_radius=0.16)
  t{tag}_k1 = field.tube@1(points=[[0, 3.4, 0], [0.9, 4.6, 0.4]], radius=0.11, end_radius=0.05)
  t{tag}_k2 = field.tube@1(points=[[0, 3.6, 0], [-0.9, 4.7, -0.5]], radius=0.11, end_radius=0.05)
  t{tag}_k3 = field.tube@1(points=[[0, 3.8, 0], [0.2, 5.0, -1.0]], radius=0.1, end_radius=0.05)
  t{tag}_ku1 = field.smooth_union@1(t{tag}_k0, t{tag}_k1, radius=0.15)
  t{tag}_ku2 = field.smooth_union@1(t{tag}_ku1, t{tag}_k2, radius=0.15)
  t{tag}_ku3 = field.smooth_union@1(t{tag}_ku2, t{tag}_k3, radius=0.15)
  t{tag}_ks = field.to_solid@1(t{tag}_ku3, min=[-1.4, -0.4, -1.4], max=[1.4, 5.3, 1.4], edge_length=0.03)
  t{tag}_kp = xform.move@1(t{tag}_ks, by=[{tx}, 0, {tz}])
  t{tag}_km = solid.material@1(t{tag}_kp, color=[0.42, 0.37, 0.3], roughness=0.9)
  t{tag}_kf = solid.finish@1(t{tag}_km, relief=0.8, relief_scale=10, relief_style="ridged")
  output t{tag}_trunk = t{tag}_kf
  t{tag}_c0 = field.ellipsoid@1(radii=[1.9, 1.4, 1.9])
  t{tag}_c0m = field.move@1(t{tag}_c0, by=[0, 6.0, 0])
  t{tag}_c1 = field.ellipsoid@1(radii=[1.4, 1.1, 1.4])
  t{tag}_c1m = field.move@1(t{tag}_c1, by=[1.5, 5.4, 0.7])
  t{tag}_c2 = field.ellipsoid@1(radii=[1.4, 1.1, 1.4])
  t{tag}_c2m = field.move@1(t{tag}_c2, by=[-1.5, 5.5, -0.6])
  t{tag}_c3 = field.ellipsoid@1(radii=[1.3, 1.0, 1.3])
  t{tag}_c3m = field.move@1(t{tag}_c3, by=[0.3, 5.3, -1.6])
  t{tag}_c4 = field.ellipsoid@1(radii=[1.2, 1.0, 1.2])
  t{tag}_c4m = field.move@1(t{tag}_c4, by=[-0.2, 5.5, 1.6])
  t{tag}_cu1 = field.smooth_union@1(t{tag}_c0m, t{tag}_c1m, radius=0.7)
  t{tag}_cu2 = field.smooth_union@1(t{tag}_cu1, t{tag}_c2m, radius=0.7)
  t{tag}_cu3 = field.smooth_union@1(t{tag}_cu2, t{tag}_c3m, radius=0.7)
  t{tag}_cu4 = field.smooth_union@1(t{tag}_cu3, t{tag}_c4m, radius=0.7)
