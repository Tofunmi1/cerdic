# A PUMP-ACTION SHOTGUN, built only with the language's own nodes: hollow lathed barrel with rib and bead, magazine tube with cap, receiver cut by booleans
# (ejection port, loading slot), grooved pump forearm, swept and rounded wooden stock, trigger group, and a fire / pump / eject cycle (4 s loop):
# recoil, muzzle flash and smoke, the forearm and bolt cycling, a spent shell tumbling out. Metres; the muzzle points to +z, y is up; the port is on the shooter's right (-x).

# ---------- materials ----------

def shellpart(tag)
  sw_{tag} = anim.track@1(s{tag}, property="position", keys="0:0,0,0; 1.3:0,0,0; 1.4:-0.05,0.05,-0.02; 1.7:-0.2,-0.04,-0.08; 2.4:-0.34,-0.55,-0.14; 4:-0.34,-0.55,-0.14", interpolation="smooth")
  sx_{tag} = anim.track@1(sw_{tag}, property="rotation", keys="0:0,0,0; 1.3:0,0,0; 2.4:230,70,30; 4:230,70,30", interpolation="linear", pivot=[-0.002, 0.008, 0.02])
  sv_{tag} = anim.track@1(sx_{tag}, property="visible", keys="0:0; 1.3:1; 2.4:0; 4:0", interpolation="step")
  sq_{tag} = anim.parent@1(sv_{tag}, parent="gun")
  output shell_{tag} = sq_{tag}
end
