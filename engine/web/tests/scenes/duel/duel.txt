# A scripted duel: two fighters fight with the combat library (attack timelines, combos, hitstop, dodge, block and parry, poise) and the game UI
# (health, stamina and poise bars drawn from procedural fonts), all driven by game.script (JavaScript inside the fixed step).
sky = scene.sky@1(sun_elevation=35, sun_azimuth=40, ground_color=[0.2, 0.22, 0.25])
ambient = scene.environment@1(sky_color=[0.6, 0.72, 0.9], ground_color=[0.25, 0.27, 0.3], intensity=0.5)
floor_box = solid.box@1(size=[14, 0.2, 8])
floor_at = xform.move@1(floor_box, by=[0, -0.1, 0])
floor = solid.material@1(floor_at, color=[0.32, 0.27, 0.22], roughness=0.9, recipe="planks")
output floor = floor

def fighter(tag, r, g, b, h, w)
  f{tag}_body = solid.box@1(size=[{w}, {h}, {w * 0.6}])
  f{tag}_up = xform.move@1(f{tag}_body, by=[0, {h * 0.5 + 0.35}, 0])
  f{tag}_head = solid.sphere@1(radius={w * 0.32}, segments=24)
  f{tag}_hup = xform.move@1(f{tag}_head, by=[0, {h + 0.35 + w * 0.3}, 0])
  f{tag}_legs = solid.box@1(size=[{w * 0.8}, 0.35, {w * 0.5}])
  f{tag}_lup = xform.move@1(f{tag}_legs, by=[0, 0.175, 0])
  f{tag}_u1 = solid.union@1(f{tag}_up, f{tag}_hup)
  f{tag}_u2 = solid.union@1(f{tag}_u1, f{tag}_lup)
  f{tag}_m = solid.material@1(f{tag}_u2, color=[{r}, {g}, {b}], roughness=0.5, metallic=0.3)
  output {tag} = f{tag}_m
end
fighter("knight", 0.25, 0.4, 0.85, 1.0, 0.7)
fighter("brute", 0.8, 0.3, 0.2, 1.3, 1.0)

state.unblock = {}; state.trail = [100, 160]; }); on('tick', (e) => { const [a, b] = state.ids.map(entity); [[a, b, 2.4, 'slash', 0.03], [b, a, 1.8, 'smash', 0.02]].forEach(([u, t, speed, heavy, rate]) => { if (rpg.is_dead(u) || rpg.is_dead(t)) return; if (combat.busy(t) && !state.unblock[u.id] && r < 0.05) { combat.input(u, 'block_on'); state.unblock[u.id] = e.tick + 30; } else if (r < rate) combat.input(u, heavy); else if (r < rate * 2.2) combat.input(u, 'slash'); else if (r < rate * 2.5) combat.input(u, 'roll'); }); const th = (id, v, i) => { state.trail[i] = Math.max(v, state.trail[i] - 0.4); return state.trail[i]; }; [[a, 'tl', 'KNIGHT', 0], [b, 'tr', 'BRUTE', 1]].forEach(([u, anchor, name, i]) => { const hp = rpg.get(u, 'health'); }); if (rpg.is_dead(a) || rpg.is_dead(b)) ui.text('banner', rpg.is_dead(b) ? 'KNIGHT WINS' : 'BRUTE WINS', { anchor: 'c', size: 0.09, font: 'bold', color: 'rgb(255,215,106)', plate: true }); });")
