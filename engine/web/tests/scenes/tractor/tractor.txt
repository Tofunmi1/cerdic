# ---- wheels: built at the origin with the axle on Y, turned to Z, moved, then spun about their own centre ----
def rear_wheel(tag, wx, wy, wz, flip)
  w{tag}_tp = profile.polygon@1(points=[[0.50, -0.27], [0.70, -0.285], [0.82, -0.25], [0.85, -0.18], [0.855, 0], [0.85, 0.18], [0.82, 0.25], [0.70, 0.285], [0.50, 0.27], [0.48, 0]])
  w{tag}_tps = profile.smooth@1(w{tag}_tp, iterations=2)
  w{tag}_tire = solid.revolve@1(w{tag}_tps, angle=360deg, segments=80)
  w{tag}_tm = solid.material@1(w{tag}_tire, color=[0.025, 0.025, 0.027], roughness=0.92)
  w{tag}_tf = solid.finish@1(w{tag}_tm, relief=0.5, relief_scale=60, relief_style="ridged", sheen=0.2)
  w{tag}_tr = xform.rotate@1(w{tag}_tf, angles=[90, {flip}, 0])
  w{tag}_ta = xform.move@1(w{tag}_tr, by=[{wx}, {wy}, {wz}])
  w{tag}_ts = anim.track@1(w{tag}_ta, property="rotation", keys="${spin_rear}", interpolation="linear", loop=true, pivot=[{wx}, {wy}, {wz}])
  output w{tag}_tire = w{tag}_ts
  w{tag}_lug = solid.box@1(size=[0.1, 0.2, 0.07])
  w{tag}_lua = xform.rotate@1(w{tag}_lug, angles=[24, 0, 0])
  w{tag}_lub = xform.move@1(w{tag}_lua, by=[0, 0.14, 0])
  w{tag}_lc = xform.rotate@1(w{tag}_lug, angles=[-24, 0, 0])
  w{tag}_ld = xform.move@1(w{tag}_lc, by=[0, -0.14, 0])
  w{tag}_la = inst.radial@1(w{tag}_lub, count=26, radius=0.88)
  w{tag}_lb = inst.radial@1(w{tag}_ld, count=26, radius=0.88)
  w{tag}_lar = inst.realize@1(w{tag}_la)
  w{tag}_lbr = inst.realize@1(w{tag}_lb)
  w{tag}_ls = solid.union@1(w{tag}_lar, w{tag}_lbr)
  w{tag}_lm = solid.material@1(w{tag}_ls, color=[0.025, 0.025, 0.027], roughness=0.92)
  w{tag}_lr = xform.rotate@1(w{tag}_lm, angles=[90, {flip}, 0])
  w{tag}_lt = xform.move@1(w{tag}_lr, by=[{wx}, {wy}, {wz}])
  w{tag}_lsn = anim.track@1(w{tag}_lt, property="rotation", keys="${spin_rear}", interpolation="linear", loop=true, pivot=[{wx}, {wy}, {wz}])
  output w{tag}_lugs = w{tag}_lsn
  w{tag}_rp = profile.polygon@1(points=[[0.40, -0.21], [0.52, -0.21], [0.52, 0.21], [0.40, 0.21]])
  w{tag}_rim = solid.revolve@1(w{tag}_rp, angle=360deg, segments=64)
  w{tag}_disc = solid.cylinder@1(radius=0.46, height=0.05, segments=48)
  w{tag}_disc_at = xform.move@1(w{tag}_disc, by=[0, 0.12, 0])
  w{tag}_hub = solid.cylinder@1(radius=0.15, height=0.34, segments=32)
  w{tag}_hub_at = xform.move@1(w{tag}_hub, by=[0, 0.1, 0])
  w{tag}_ru = solid.union@1(w{tag}_rim, w{tag}_disc_at, w{tag}_hub_at)
  w{tag}_rm = solid.material@1(w{tag}_ru, color=[0.92, 0.72, 0.08], metallic=0.5, roughness=0.4)
  w{tag}_rt = solid.finish@1(w{tag}_rm, clearcoat=0.5, clearcoat_roughness=0.15)
  w{tag}_rr = xform.rotate@1(w{tag}_rt, angles=[90, {flip}, 0])
  w{tag}_ra = xform.move@1(w{tag}_rr, by=[{wx}, {wy}, {wz}])
  w{tag}_rs = anim.track@1(w{tag}_ra, property="rotation", keys="${spin_rear}", interpolation="linear", loop=true, pivot=[{wx}, {wy}, {wz}])
  output w{tag}_rim = w{tag}_rs
  w{tag}_bolt = solid.cylinder@1(radius=0.02, height=0.07, segments=10)
  w{tag}_bolt_at = xform.move@1(w{tag}_bolt, by=[0, 0.16, 0])
  w{tag}_bolts = inst.radial@1(w{tag}_bolt_at, count=8, radius=0.1)
  w{tag}_bm = solid.material@1(w{tag}_bolts, color=[0.5, 0.52, 0.55], metallic=0.95, roughness=0.3)
  w{tag}_br = xform.rotate@1(w{tag}_bm, angles=[90, {flip}, 0])
  w{tag}_bt = xform.move@1(w{tag}_br, by=[{wx}, {wy}, {wz}])
  w{tag}_bs = anim.track@1(w{tag}_bt, property="rotation", keys="${spin_rear}", interpolation="linear", loop=true, pivot=[{wx}, {wy}, {wz}])
  output w{tag}_bolts = w{tag}_bs
end
