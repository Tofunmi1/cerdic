# one bone: a part that turns about its own joint, with a clip for every move (a clip the bone does not use is `still`)
def limb(name, parent, idle, run, slash, slash2, heavy, roll, hit, block, cast, death)
  hb_{name} = body.bone@1(hbody, bone={name})
  hp_{name} = anim.parent@1(hb_{name}, parent={parent})
  hc_{name}_0 = anim.track@1(hp_{name}, property="rotation", keys={idle}, interpolation="smooth", clip="idle", pivot_at="head")
  hc_{name}_1 = anim.track@1(hc_{name}_0, property="rotation", keys={run}, interpolation="smooth", clip="run", pivot_at="head")
  hc_{name}_2 = anim.track@1(hc_{name}_1, property="rotation", keys={slash}, interpolation="smooth", loop=false, clip="slash", pivot_at="head")
  hc_{name}_3 = anim.track@1(hc_{name}_2, property="rotation", keys={slash2}, interpolation="smooth", loop=false, clip="slash2", pivot_at="head")
  hc_{name}_4 = anim.track@1(hc_{name}_3, property="rotation", keys={heavy}, interpolation="smooth", loop=false, clip="heavy", pivot_at="head")
  hc_{name}_5 = anim.track@1(hc_{name}_4, property="rotation", keys={roll}, interpolation="smooth", loop=false, clip="roll", pivot_at="head")
  hc_{name}_6 = anim.track@1(hc_{name}_5, property="rotation", keys={hit}, interpolation="smooth", loop=false, clip="hit", pivot_at="head")
  hc_{name}_7 = anim.track@1(hc_{name}_6, property="rotation", keys={block}, interpolation="smooth", loop=false, clip="block", pivot_at="head")
  hc_{name}_8 = anim.track@1(hc_{name}_7, property="rotation", keys={cast}, interpolation="smooth", loop=false, clip="cast", pivot_at="head")
  hc_{name}_9 = anim.track@1(hc_{name}_8, property="rotation", keys={death}, interpolation="smooth", loop=false, clip="death", pivot_at="head")
  output h_{name} = hc_{name}_9
end

# eyes: a white ball, a coloured iris cap and a dark pupil, each a part on the head bone (the library's own eyes did not follow the head in play)
def eyepart(tag, r, z, cr, cg, cb, shine)
  he_{tag}_s = solid.sphere@1(radius={r}, segments=20)
  he_{tag}_a = xform.move@1(he_{tag}_s, by=[0.0315, 1.687, {z}])
  he_{tag}_b = xform.mirror@1(he_{tag}_a, axis="x")
  he_{tag}_k = body.skin_to@1(he_{tag}_b, hbody, bones="${hbones}", prefix="h_", blend="linear")
  he_{tag}_m = solid.material@1(he_{tag}_k, color=[{cr}, {cg}, {cb}], roughness={shine})
  output h_eye_{tag} = he_{tag}_m
end
