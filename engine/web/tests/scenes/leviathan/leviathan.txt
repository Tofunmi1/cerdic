# A joint bone, and a bone that swings (about Z, a nod of the spine) with its own phase.
def bw(tag, hx, hy, hz, tx, ty, tz, par, az, per, ph)
  bw{tag}_a = solid.box@1(size=[0.02, 0.02, 0.02])
  bw{tag}_b = xform.move@1(bw{tag}_a, by=[{hx}, {hy}, {hz}])
  bw{tag}_c = rig.bone@1(bw{tag}_b, head=[{hx}, {hy}, {hz}], tail=[{tx}, {ty}, {tz}])
  bw{tag}_d = anim.parent@1(bw{tag}_c, parent="${par}")
  bw{tag}_e = anim.wave@1(bw{tag}_d, property="rotation", amplitude=[0, 0, {az}], period={per}, phase={ph}, pivot_at="head")
  output {tag} = bw{tag}_e
end

# ---- sails: a triangular grid of vertices with a billow, scripted, double-sided, canvas; they flutter about the mast ----
def sail(tag, ax, ay, bx, by, cx, cy, bil, per, ph)
  sl{tag}_s = poly.script@1(code="const m = new Mesh(); const N = 12; const A = [${ax}, ${ay}, 0], B = [${bx}, ${by}, 0], C = [${cx}, ${cy}, 0]; const V = []; for (let i = 0; i <= N; i++) { const row = []; for (let j = 0; j <= N - i; j++) { const u = i / N, v = j / N, w = 1 - u - v; const z = ${bil} * 27 * u * v * w; row.push(m.vertex(A[0] + u * (B[0] - A[0]) + v * (C[0] - A[0]), A[1] + u * (B[1] - A[1]) + v * (C[1] - A[1]), z)); } V.push(row); } for (let i = 0; i < N; i++) { for (let j = 0; j <= N - i - 1; j++) { m.face(V[i][j], V[i + 1][j], V[i][j + 1]); m.face(V[i][j], V[i][j + 1], V[i + 1][j]); if (j <= N - i - 2) { m.face(V[i + 1][j], V[i + 1][j + 1], V[i][j + 1]); m.face(V[i + 1][j], V[i][j + 1], V[i + 1][j + 1]); } } } m", budget=2000)
  sl{tag}_m = solid.material@1(sl{tag}_s, color=[0.93, 0.88, 0.76], roughness=0.9, recipe="fabric", texture_scale=14)

# ---- two meshing brass gears on the cabin wall: toothed discs made by a script, the big one turning slowly, the small one faster the other way ----
def gear(tag, gr, gt, gx, keys)
  gr{tag}_s = poly.script@1(code="const m = new Mesh(); const R = ${gr}, T = ${gt}, th = 0.06; const rin = R * 0.62, root = R * 0.86, p = 2 * Math.PI / T; const out = []; for (let i = 0; i < T; i++) { const a = i * p; out.push([root, a], [root, a + 0.2 * p], [R, a + 0.32 * p], [R, a + 0.6 * p], [root, a + 0.72 * p]); } const n = out.length; const fo = [], bo = [], fi = [], bi = []; for (let k = 0; k < n; k++) { const r = out[k][0], a = out[k][1]; fo.push(m.vertex(r * Math.cos(a), r * Math.sin(a), th / 2)); bo.push(m.vertex(r * Math.cos(a), r * Math.sin(a), -th / 2)); fi.push(m.vertex(rin * Math.cos(a), rin * Math.sin(a), th / 2)); bi.push(m.vertex(rin * Math.cos(a), rin * Math.sin(a), -th / 2)); } const fc = m.vertex(0, 0, th / 2), bc = m.vertex(0, 0, -th / 2); for (let k = 0; k < n; k++) { const j = (k + 1) % n; m.face(fi[k], fo[k], fo[j], fi[j]); m.face(bi[k], bi[j], bo[j], bo[k]); m.face(fo[k], bo[k], bo[j], fo[j]); m.face(fc, fi[k], fi[j]); m.face(bc, bi[j], bi[k]); } m", budget=2000)
  gr{tag}_m = solid.material@1(gr{tag}_s, color=[0.8, 0.55, 0.18], metallic=1.0, roughness=0.32)
  gr{tag}_f = solid.finish@1(gr{tag}_m, clearcoat=0.4, clearcoat_roughness=0.25, relief=0.3, relief_scale=30, relief_style="ridged")
  gr{tag}_a = xform.move@1(gr{tag}_f, by=[{gx}, 5.7, -0.53])
  gr{tag}_p = anim.parent@1(gr{tag}_a, parent="b2")
  gr{tag}_t = anim.track@1(gr{tag}_p, property="rotation", keys="${keys}", interpolation="linear", loop=true, pivot=[{gx}, 5.7, -0.53])
  output gear_{tag} = gr{tag}_t
end
gear("big", 0.5, 16, 0.3, "0:0,0,0; 2:0,0,90; 4:0,0,180; 6:0,0,270; 8:0,0,360")
gear("small", 0.3, 10, 1.04, "0:0,0,0; 1.25:0,0,-90; 2.5:0,0,-180; 3.75:0,0,-270; 5:0,0,-360")

# ---- rigging: a rope swept along a Bezier with a string of lantern bulbs hanging along it ----
def rope(tag, x0, y0, z0, x1, y1, z1, d)
  rp{tag}_c = curve.bezier@1(points=[[{x0}, {y0}, {z0}], [{x0 + (x1 - x0) / 3}, {y0 + (y1 - y0) / 3 - d}, {z0 + (z1 - z0) / 3}], [{x0 + (x1 - x0) * 2 / 3}, {y0 + (y1 - y0) * 2 / 3 - d}, {z0 + (z1 - z0) * 2 / 3}], [{x1}, {y1}, {z1}]], steps=20)
  rp{tag}_r = profile.circle@1(radius=0.018, segments=8)
  rp{tag}_s = solid.sweep@1(rp{tag}_c, rp{tag}_r, scale_end=1.0)
  rp{tag}_m = solid.material@1(rp{tag}_s, color=[0.55, 0.42, 0.25], roughness=0.9)
  rp{tag}_k = anim.parent@1(rp{tag}_m, parent="b2")
  output rope_{tag} = rp{tag}_k
  rp{tag}_b = solid.sphere@1(radius=0.075, segments=16)
  rp{tag}_bp = inst.place@1(rp{tag}_b, at=[[{x0 + (x1 - x0) * 0.2}, {y0 + (y1 - y0) * 0.2 - 3 * d * 0.16 - 0.1}, {z0 + (z1 - z0) * 0.2}], [{x0 + (x1 - x0) * 0.35}, {y0 + (y1 - y0) * 0.35 - 3 * d * 0.2275 - 0.1}, {z0 + (z1 - z0) * 0.35}], [{x0 + (x1 - x0) * 0.5}, {y0 + (y1 - y0) * 0.5 - 3 * d * 0.25 - 0.1}, {z0 + (z1 - z0) * 0.5}], [{x0 + (x1 - x0) * 0.65}, {y0 + (y1 - y0) * 0.65 - 3 * d * 0.2275 - 0.1}, {z0 + (z1 - z0) * 0.65}], [{x0 + (x1 - x0) * 0.8}, {y0 + (y1 - y0) * 0.8 - 3 * d * 0.16 - 0.1}, {z0 + (z1 - z0) * 0.8}]])
  rp{tag}_bm = solid.material@1(rp{tag}_bp, color=[1, 0.82, 0.45], roughness=0.15, emissive=[3.2, 2.0, 0.7])
  rp{tag}_bk = anim.parent@1(rp{tag}_bm, parent="b2")
  rp{tag}_bg = anim.wave@1(rp{tag}_bk, property="glow", amplitude=[0.25, 0, 0], period=1.7, phase=0, phase_step=0.23)
  output bulbs_{tag} = rp{tag}_bg
end
rope("bow", 0.65, 8.85, 0, 2.2, 6.0, 0, 0.35)
rope("stern", 0.65, 8.85, 0, -1.75, 5.9, 0, 0.35)
bulb_light_a = light.point@1(position=[1.4, 7.0, 0.15], color=[1, 0.8, 0.45], intensity=40, range=4)
bulb_light_b = light.point@1(position=[-0.7, 7.0, 0.15], color=[1, 0.8, 0.45], intensity=40, range=4)
output lantern_light_a = bulb_light_a
output lantern_light_b = bulb_light_b

# ---- girth bands: brass hoops round the whale with rivets, tying the gondola on ----
band = solid.torus@1(major=1.76, minor=0.07, segments=64, tube_segments=14)
band_x = xform.rotate@1(band, angles=[0, 0, 90])
band_a = xform.move@1(band_x, by=[0.9, 3.2, 0])
band_b = xform.move@1(band_x, by=[-0.6, 3.2, 0])
band_am = solid.material@1(band_a, color=[0.8, 0.55, 0.18], metallic=1.0, roughness=0.3)
band_bm = solid.material@1(band_b, color=[0.8, 0.55, 0.18], metallic=1.0, roughness=0.3)
band_ak = anim.parent@1(band_am, parent="b1")
band_bk = anim.parent@1(band_bm, parent="b2")
output band_front = band_ak
output band_rear = band_bk
