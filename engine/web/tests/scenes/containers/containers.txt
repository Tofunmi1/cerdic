# a part: a builder call, a colour, metallic and roughness
def part(name, call, r, g, b, met, rgh, rec)
  {name}_s = poly.script@1(code="${jc} ${jx} ${call}")
  {name}_m = solid.material@1(${name}_s, color=[{r}, {g}, {b}], metallic={met}, roughness={rgh}, recipe="${rec}")
  output {name} = ${name}_m
end
def pglow(name, call, r, g, b, er, eg, eb)
  {name}_s = poly.script@1(code="${jc} ${jx} ${call}")
  {name}_m = solid.material@1(${name}_s, color=[{r}, {g}, {b}], roughness=0.3, emissive=[{er}, {eg}, {eb}])
  output {name} = ${name}_m
end

# weathered steel, not plastic: the paint fades and streaks (noise, stretched vertically like rain runs), rust gathers in the corrugation valleys (cavity), bare steel shows on the ridges and edges (convexity),
# dirt darkens the creases (ambient occlusion), and roughness and metal follow the wear (shiny bare steel, matt rusty and dusty paint). The two hex colours are the dark and light ends of the paint (no # in a call).
def ctr(name, call, r, g, b, c0, c1)
  {name}_s = poly.script@1(code="${jc} ${jx} ${call}")
  {name}_m = solid.material@1(${name}_s, color=[{r}, {g}, {b}], metallic=0.3, roughness=0.6)
