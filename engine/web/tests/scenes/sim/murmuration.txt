# MURMURATION: 300 starlings flying by three local rules (stay close, match your neighbours' heading, do not collide) while a slow roost point
# drifts across the sky and pulls the whole flock after it. There is no flocking node: the rules are a SimL program in `source`, run by the
# engine's simulation substrate (sim.program), and the same text could drive cloth, hair or a crowd. Metres, y up.
sky = scene.sky@1(sun_elevation=7, sun_azimuth=-60, sun_intensity=1.0, turbidity=3.5, ground_color=[0.05, 0.05, 0.06], clouds=true, haze=0.0)
output sky = sky
env = scene.environment@1(sky_color=[0.75, 0.5, 0.35], ground_color=[0.08, 0.07, 0.07], intensity=0.7, reflection=0.3)
output env = env

ground_s = solid.box@1(size=[200, 0.5, 200])
ground_m = solid.material@1(ground_s, color=[0.09, 0.1, 0.08], roughness=0.95)
output ground = ground_m

flock = sim.program@1(size=0.45, seed=11, rate=30, source="sim murmuration\nparam radius = 7.0\nparam max_speed = 11.0\nparam cohesion = 0.9\nparam separation = 4.0\nparam align = 3.5\nparam roost = 5.0\nstate pos: vec3[300]\nstate vel: vec3[300]\nkernel init each i in pos {\n  pos[i] = vec3(rand(i, 1.0), rand(i, 2.0), rand(i, 3.0)) * 16.0 - 8.0 + vec3(0.0, 15.0, 0.0)\n  vel[i] = normalize(vec3(rand(i, 4.0), rand(i, 5.0), rand(i, 6.0)) - 0.5) * 8.0\n}\nkernel flock each i in pos {\n  let p = pos[i]\n  let v = vel[i]\n  let centre = vec3(0.0)\n  let heading = vec3(0.0)\n  let push = vec3(0.0)\n  let seen = 0.0\n  for j in neighbors(pos, p, radius) cap 16 {\n    if j != i {\n      centre += pos[j]\n      heading += vel[j]\n      let d = p - pos[j]\n      push += d / max(dot(d, d), 0.05)\n      seen += 1.0\n    }\n  }\n  let acc = vec3(0.0)\n  if seen > 0.0 {\n    acc = (centre / seen - p) * cohesion + (heading / seen - v) * align + push * separation\n  }\n  let target = vec3(sin(t * 0.21) * 30.0, 16.0 + sin(t * 0.37) * 5.0, cos(t * 0.17) * 30.0)\n  acc += normalize(target - p) * roost\n  if p.y < 7.0 {\n    acc += vec3(0.0, (7.0 - p.y) * 3.0, 0.0)\n  }\n  let nv = v + acc * dt\n  let sp = length(nv)\n  nv = nv * min(1.0, max_speed / max(sp, 0.001))\n  nv = nv * max(1.0, 6.0 / max(sp, 0.001))\n  vel[i] = nv\n}\nkernel move each i in pos { pos[i] += vel[i] * dt }\nschedule flock; move\noutput particles pos\n")
birds = solid.material@1(flock, color=[0.03, 0.03, 0.04], roughness=0.8)
output birds = birds
