# Mechanisms demo. Press Play. Everything here hangs on joints:
#  - a pendulum (rod hinged to the ceiling bar, a bob welded to its end) swinging from the horizontal,
#  - a door on a hinge and a drawer on a slider: drag them with the mouse,
#  - a weight on a spring, bouncing,
#  - a windmill turned by a hinge motor,
#  - a plank bridge (six planks hinged to each other and to two posts) with a heavy ball on it,
#  - a seesaw with a ball dropped on one end,
#  - a bell: the ball that falls on it sets the flag "ring", and the lamp beside it lights.
# Drag any loose part to move or throw it. Stop puts everything back.

floor_box = solid.box@1(size=[16, 0.2, 14])
floor_down = xform.move@1(floor_box, by=[0, -0.1, 1.5])
floor_tint = solid.material@1(floor_down, color=[0.42, 0.47, 0.42], roughness=0.9)
floor = phys.body@1(floor_tint, kind="static", shape="box", friction=0.7)
output ground = floor

# Low walls so loose balls stay on the floor.
wall_long = solid.box@1(size=[16.4, 0.6, 0.2])

wall_tint = solid.material@1(wall_long, color=[0.55, 0.58, 0.7], roughness=0.8)
wall_short_tint = solid.material@1(wall_short, color=[0.55, 0.58, 0.7], roughness=0.8)
wall_n_at = xform.move@1(wall_tint, by=[0, 0.3, -5.6])
wall_n = phys.body@1(wall_n_at, kind="static", shape="box")
output wall_north = wall_n
wall_s_at = xform.move@1(wall_tint, by=[0, 0.3, 8.6])
wall_s = phys.body@1(wall_s_at, kind="static", shape="box")
output wall_south = wall_s
wall_e_at = xform.move@1(wall_short_tint, by=[8.1, 0.3, 1.5])
wall_e = phys.body@1(wall_e_at, kind="static", shape="box")
output wall_east = wall_e
wall_w_at = xform.move@1(wall_short_tint, by=[-8.1, 0.3, 1.5])
wall_w = phys.body@1(wall_w_at, kind="static", shape="box")
output wall_west = wall_w

# Pendulum.
bar = solid.box@1(size=[3, 0.1, 0.1])
bar_at = xform.move@1(bar, by=[-5.2, 4.4, -3])
bar_steel = solid.material@1(bar_at, color=[0.5, 0.55, 0.6], roughness=0.4)
ceiling_bar = phys.body@1(bar_steel, kind="static", shape="box")
output pendulum_bar = ceiling_bar
rod = solid.box@1(size=[1.5, 0.05, 0.05])
rod_at = xform.move@1(rod, by=[-5.25, 4, -3])
rod_steel = solid.material@1(rod_at, color=[0.7, 0.7, 0.75], roughness=0.3)
rod_body = phys.body@1(rod_steel, shape="box", density=30)
rod_hinge = phys.joint@1(rod_body, kind="hinge", to="pendulum_bar", anchor=[-6, 4, -3], axis=[0, 0, 1], damping=0.02)
output pendulum_rod = rod_hinge
bob = solid.sphere@1(radius=0.3, segments=24)
bob_red = solid.material@1(bob, color=[0.85, 0.15, 0.15], roughness=0.3)
bob_at = xform.move@1(bob_red, by=[-4.5, 4, -3])
bob_body = phys.body@1(bob_at, shape="sphere", density=4)
bob_weld = phys.joint@1(bob_body, kind="fixed", to="pendulum_rod", anchor=[-4.5, 4, -3])
output pendulum_bob = bob_weld

# Windmill: the blades turn on a hinge with a motor.
tower = solid.box@1(size=[0.3, 3, 0.3])
tower_at = xform.move@1(tower, by=[-6, 1.5, 3])
tower_stone = solid.material@1(tower_at, color=[0.75, 0.7, 0.6], roughness=0.9)
windmill_tower = phys.body@1(tower_stone, kind="static", shape="box")
output windmill_tower = windmill_tower
blades = solid.box@1(size=[3, 0.25, 0.05])
blades_at = xform.move@1(blades, by=[-6, 3.1, 3.25])
blades_cross = solid.box@1(size=[0.25, 3, 0.05])
blades_cross_at = xform.move@1(blades_cross, by=[-6, 3.1, 3.25])
blades_both = solid.union@1(blades_at, blades_cross_at)
blades_red = solid.material@1(blades_both, color=[0.8, 0.25, 0.2], roughness=0.5)
blades_body = phys.body@1(blades_red, shape="convex", density=0.5)
blades_motor = phys.joint@1(blades_body, kind="hinge", to="windmill_tower", anchor=[-6, 3.1, 3.25], axis=[0, 0, 1], motor_speed=80, motor_force=30)
output windmill_blades = blades_motor

# A bell, and a ball that rings it: the collision sets the flag "ring", and the lamp follows that flag.
bell = solid.sphere@1(radius=0.3, segments=24)
bell_gold = solid.material@1(bell, color=[0.95, 0.75, 0.2], roughness=0.25, metallic=0.8)
bell_at = xform.move@1(bell_gold, by=[3, 1.4, 6])
bell_body = phys.body@1(bell_at, kind="static", shape="sphere")
output bell = bell_body
clapper = solid.sphere@1(radius=0.18, segments=24)
clapper_blue = solid.material@1(clapper, color=[0.2, 0.4, 0.95], roughness=0.3)
clapper_at = xform.move@1(clapper_blue, by=[3.1, 4, 6])
clapper_body = phys.body@1(clapper_at, shape="sphere", restitution=0.5)
clapper_rings = behavior.on@1(clapper_body, trigger="collide", accepts="bell", sets="ring")
output ringer = clapper_rings
lamp_bulb = solid.sphere@1(radius=0.2, segments=24)
lamp_glass = solid.material@1(lamp_bulb, color=[0.05, 0.05, 0.05], emissive=[3, 2.4, 1.2], emits_light=true, roughness=0.3)
lamp_at = xform.move@1(lamp_glass, by=[4.6, 1.4, 6])
lamp_glow = anim.track@1(lamp_at, property="glow", keys="0:0; 0.4:1", loop=false)
lamp_follows = behavior.on@1(lamp_glow, trigger="flag", watch="ring", action="follow")
output lamp = lamp_follows
