# Moon walk. Press Play, then click the view to capture the mouse: WASD walks, Shift runs, Space jumps, the mouse looks.
# Gravity here is the Moon's (1.62 m/s2), so jumps are long and floaty. You stand on the floor, can hop up the four
# platforms to the front (0.6 to 2.4 m high), walk up the stairs and the ramp at the side, and shove the crates around
# (they are loose bodies: drag one with the mouse, or run into the stack and watch it fall slowly).
# Press Stop to go back to the flying camera.

floor_box = solid.box@1(size=[40, 0.2, 40])
floor_down = xform.move@1(floor_box, by=[0, -0.1, 0])
floor_dust = solid.material@1(floor_down, color=[0.55, 0.53, 0.5], roughness=0.95)
floor_body = phys.body@1(floor_dust, kind="static", shape="box", friction=0.9)
floor = phys.world@1(floor_body, gravity=[0, -1.62, 0], walker=true, jump=3.2, speed=3.5)
output ground = floor
