# Village in the afternoon. Press Play, click the view, then WASD walks, Shift runs, Space jumps, the mouse looks.
# Brick house with a hinged door (drag it with the mouse), a stone pond with moving water, a marble well, lamp posts,
# six trees, a fence, and loose crates and a ball to knock about. Everything is procedural: recipes for brick,
# planks, wood, marble and metal, a texture program for the grass, noise for the tree canopies, water and sky nodes.

sunset = scene.sky@1(sun_elevation=27, sun_azimuth=-40, sun_intensity=6, turbidity=3, ground_color=[0.45, 0.38, 0.28], clouds=true, cloud_cover=0.4, seed=21)
output sky = sunset

# Grass ground, with the scene's physics settings (walk, Earth gravity).
ground_box = solid.box@1(size=[34, 0.4, 34])
ground_low = xform.move@1(ground_box, by=[0, -0.2, 0])
ground_grass = solid.texture@1(ground_low, program="n = fbm(scale_x=9, scale_y=9); m = fbm(scale_x=30, scale_y=30); albedo = colorize(n, gradient=[0:#4d7a35, 0.5:#79a84a, 1:#b5c26a]); roughness = m")
ground_body = phys.body@1(ground_grass, kind="static", shape="box", friction=0.9)
ground = phys.world@1(ground_body, gravity=[0, -9.81, 0], walker=true, speed=3.2, jump=4.5)
output ground = ground

# Cobbled plaza.
plaza_box = solid.box@1(size=[10, 0.1, 10])

plaza = phys.body@1(plaza_stone, kind="static", shape="box", friction=0.8)
output plaza = plaza

# A stone pond with water at the back right.
rim_outer = solid.cylinder@1(radius=3, height=0.45, segments=48)
rim_cut = solid.cylinder@1(radius=2.5, height=0.8, segments=48)

rim_stone = solid.material@1(rim_ring, color=[0.5, 0.5, 0.52], roughness=0.85, recipe="concrete", texture_scale=2)
rim_at = xform.move@1(rim_stone, by=[7, 0.225, -5])
pond_rim = phys.body@1(rim_at, kind="static", shape="mesh")

pond_water = water.surface@1(size=[3.4, 3.4], level=0.32, depth=0.28, color=[0.08, 0.36, 0.5], opacity=0.62, wave_height=0.02, wave_length=1.2, speed=0.8, seed=4, resolution=40)
pond_at = xform.move@1(pond_water, by=[7, 0, -5])
output pond = pond_at

# A fence along the back edge: one post-and-rail segment repeated.
fence_post = solid.box@1(size=[0.12, 1.1, 0.12])
fence_rail = solid.box@1(size=[2, 0.08, 0.06])
fence_rail_up = xform.move@1(fence_rail, by=[1, 0.35, 0])
fence_rail_high = xform.move@1(fence_rail, by=[1, 0.8, 0])
fence_posts = solid.union@1(fence_post, fence_rail_up, fence_rail_high)
fence_wood = solid.material@1(fence_posts, color=[0.55, 0.4, 0.25], roughness=0.8, recipe="planks", texture_scale=3)
fence_start = xform.move@1(fence_wood, by=[-16, 0.55, 16])
fence = inst.repeat@1(fence_start, count=16, step=[2, 0, 0])
output fence = fence
