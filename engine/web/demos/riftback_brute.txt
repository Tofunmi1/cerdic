def digit(tag, x0, y0, z0, dx, length)
  d{tag}_p = curve.spline@1(points=[[{x0}, {y0}, {z0}], [{x0 + dx * 0.42}, {y0 - length * 0.42}, {z0 + 0.025}], [{x0 + dx}, {y0 - length}, {z0 + 0.055}], [{x0 + dx * 1.08}, {y0 - length - 0.055}, {z0 + 0.095}]], steps=10)
  d{tag}_r = profile.circle@1(radius=0.067, segments=14)
  d{tag}_s = solid.sweep@1(d{tag}_p, d{tag}_r, scale_end=0.62)
  d{tag}_m = solid.material@1(d{tag}_s, color=[0.74, 0.75, 0.73], roughness=0.91)
  d{tag}_t = solid.texture@1(d{tag}_m, tile=0.8, program="${hide_program}", size="256")
  d{tag}_c = curve.spline@1(points=[[{x0 + dx * 1.08}, {y0 - length - 0.055}, {z0 + 0.095}], [{x0 + dx * 1.12}, {y0 - length - 0.14}, {z0 + 0.15}], [{x0 + dx * 1.03}, {y0 - length - 0.21}, {z0 + 0.20}]], steps=8)
  d{tag}_cr = profile.circle@1(radius=0.050, segments=12)
  d{tag}_cs = solid.sweep@1(d{tag}_c, d{tag}_cr, scale_end=0.08)
  d{tag}_cm = solid.material@1(d{tag}_cs, color=[0.12, 0.115, 0.105], metallic=0.12, roughness=0.47)
  output digit_{tag} = d{tag}_t
  output talon_{tag} = d{tag}_cm
end
