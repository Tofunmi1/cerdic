# Reusable solid furniture and architecture primitives. Every box call creates one editable output.
def cube_part(tag,sx,sy,sz,x,y,z,r,g,b,rough)
  cp_{tag}_shape = solid.box@1(size=[{sx},{sy},{sz}])
  cp_{tag}_at = xform.move@1(cp_{tag}_shape,by=[{x},{y},{z}])
  cp_{tag}_mat = solid.material@1(cp_{tag}_at,color=[{r},{g},{b}],roughness={rough})
  output {tag} = cp_{tag}_mat
end
