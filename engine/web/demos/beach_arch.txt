# Beach with a rock arch. Press Play, click the view, then WASD walks, Shift runs, Space jumps, the mouse looks.
# Everything is generated from this text: a fractal island of sand, a rippling ocean, a sandstone arch swept along a spline and
# roughened with noise, boulders, palms (swept trunks with swept fronds) dropped onto the sand, and a low evening sky.

evening = scene.sky@1(sun_elevation=34, sun_azimuth=35, sun_intensity=7, turbidity=2.4, ground_color=[0.55, 0.45, 0.32], clouds=true, cloud_cover=0.45, seed=21)
output sky = evening
fill = scene.environment@1(sky_color=[0.62, 0.74, 0.95], ground_color=[0.78, 0.66, 0.48], intensity=0.85, reflection=0.15, bounce=0.4, bounce_color=[0.9, 0.78, 0.55])
output fill = fill

# ---- Island of sand and the sea around it ----
island = terrain.heightfield@1(size=[64, 46], height=4, scale=14, octaves=4, roughness=0.45, seed=5, resolution=128, base=1, style="hills", edge_falloff=0.7, flat_below=0.2)
island_sand = solid.texture@1(island, tile=3, program="n = fbm(scale_x=2, scale_y=2); albedo = colorize(n, gradient=[0:#f0d6a2, 1:#f8e6b8]); roughness = colorize(n, gradient=[0:#e0e0e0, 1:#f0f0f0])")
island_toon = solid.toon@1(island_sand, bands=2, feather=0.07, shade_color=[0.93, 0.78, 0.72])
island_body = phys.body@1(island_toon, kind="static", shape="mesh", friction=0.9)
land = phys.world@1(island_body, gravity=[0, -9.81, 0], walker=true, speed=3.2, jump=4.5)
output land = land

sea = water.surface@1(size=[84, 62], level=0.3, depth=0.3, color=[0.04, 0.45, 0.55], opacity=0.7, wave_height=0.07, wave_length=7, speed=0.6, seed=8, resolution=96)
output sea = sea

# ---- The arch: a spline swept with a round profile, then roughened ----
arch_path = curve.spline@1(points=[[-6, 0.4, 0], [-6.3, 2.8, 0], [-4.6, 5.6, 0], [0, 7.3, 0], [4.6, 5.6, 0], [6.3, 2.8, 0], [6, 0.4, 0]], steps=10)
arch_ring = profile.circle@1(radius=1.5, segments=24)
arch_tube = solid.sweep@1(arch_path, arch_ring)
arch_rough = xform.noise@1(arch_tube, amplitude=0.5, scale=6, seed=4, tolerance=0.1)
arch_rock = solid.texture@1(arch_rough, tile=3, program="s = fbm(scale_x=2, scale_y=16); n = fbm(scale_x=7, scale_y=7); f = fbm(scale_x=30, scale_y=30); albedo = colorize(s, gradient=[0:#9a6f4a, 0.4:#b58a5c, 0.7:#c9a273, 1:#a37a52]); roughness = colorize(f, gradient=[0:#b8b8b8, 1:#e0e0e0]); height = colorize(f, gradient=[0:#808080, 1:#9a9a9a])")
arch_toon = solid.toon@1(arch_rock, bands=2, feather=0.07, shade_color=[0.72, 0.58, 0.78], rim=0.25, outline=2.5, outline_color=[0.28, 0.14, 0.12])
output arch = arch_toon

foot_left_ball = solid.sphere@1(radius=2.4, segments=24)
foot_left_rough = xform.noise@1(foot_left_ball, amplitude=0.4, scale=5, seed=11, tolerance=0.1)
foot_left_rock = solid.texture@1(foot_left_rough, tile=3, program="s = fbm(scale_x=2, scale_y=16); n = fbm(scale_x=7, scale_y=7); f = fbm(scale_x=30, scale_y=30); albedo = colorize(s, gradient=[0:#9a6f4a, 0.4:#b58a5c, 0.7:#c9a273, 1:#a37a52]); roughness = colorize(f, gradient=[0:#b8b8b8, 1:#e0e0e0]); height = colorize(f, gradient=[0:#808080, 1:#9a9a9a])")
foot_left_toon = solid.toon@1(foot_left_rock, bands=2, feather=0.07, shade_color=[0.72, 0.58, 0.78], rim=0.2, outline=2.5, outline_color=[0.28, 0.14, 0.12])
foot_left_at = xform.move@1(foot_left_toon, by=[-6, 0.4, 0.2])
output foot_left = foot_left_at

foot_right_ball = solid.sphere@1(radius=2.2, segments=24)
foot_right_rough = xform.noise@1(foot_right_ball, amplitude=0.4, scale=5, seed=17, tolerance=0.1)
foot_right_rock = solid.texture@1(foot_right_rough, tile=3, program="s = fbm(scale_x=2, scale_y=16); n = fbm(scale_x=7, scale_y=7); f = fbm(scale_x=30, scale_y=30); albedo = colorize(s, gradient=[0:#9a6f4a, 0.4:#b58a5c, 0.7:#c9a273, 1:#a37a52]); roughness = colorize(f, gradient=[0:#b8b8b8, 1:#e0e0e0]); height = colorize(f, gradient=[0:#808080, 1:#9a9a9a])")
foot_right_toon = solid.toon@1(foot_right_rock, bands=2, feather=0.07, shade_color=[0.72, 0.58, 0.78], rim=0.2, outline=2.5, outline_color=[0.28, 0.14, 0.12])
foot_right_at = xform.move@1(foot_right_toon, by=[6, 0.3, -0.2])
output foot_right = foot_right_at

# ---- A palm: swept trunk and six swept fronds; copies are scattered and dropped onto the sand ----
trunk_path = curve.spline@1(points=[[-1.6, 0, 0], [-1.3, 2, 0], [-0.7, 4, 0], [0, 5.6, 0]], steps=10)
trunk_ring = profile.circle@1(radius=0.3, segments=14)
trunk_sweep = solid.sweep@1(trunk_path, trunk_ring, scale_end=0.55)
frond_path = curve.spline@1(points=[[0, 0, 0], [1.2, 0.8, 0], [2.6, 0.5, 0], [3.7, -0.6, 0]], steps=8)
frond_ring = profile.circle@1(radius=0.4, segments=8)
frond_sweep = solid.sweep@1(frond_path, frond_ring, scale_end=0.15)
frond_flat = xform.scale@1(frond_sweep, by=[1, 0.07, 1])
frond_up = xform.move@1(frond_flat, by=[0, 5.6, 0])
palm_leaf_one = solid.texture@1(frond_up, tile=1, program="g = fbm(scale_x=3, scale_y=12); albedo = colorize(g, gradient=[0:#2f5a22, 0.5:#4c8330, 1:#7aab45]); roughness = colorize(g, gradient=[0:#a0a0a0, 1:#c8c8c8])")
palm_bark = solid.texture@1(trunk_sweep, tile=1, program="g = fbm(scale_x=3, scale_y=8); albedo = colorize(g, gradient=[0:#9a6a40, 1:#b88454]); roughness = colorize(g, gradient=[0:#e0e0e0, 1:#f0f0f0])")
palm_leaf_toon = solid.toon@1(palm_leaf_one, bands=2, feather=0.07, shade_color=[0.55, 0.7, 0.75], spec=0.3, rim=0.3, outline=1.5, outline_color=[0.08, 0.2, 0.12])
palm_bark_toon = solid.toon@1(palm_bark, bands=2, feather=0.07, shade_color=[0.7, 0.58, 0.72], outline=2, outline_color=[0.2, 0.12, 0.1])
palm_crown = inst.radial@1(palm_leaf_toon, count=8, radius=0)
trunks_spread = inst.scatter@1(palm_bark_toon, count=5, width=34, depth=18, min_spacing=6, yaw_jitter=3.14, scale_jitter=0.25, seed=9)
crowns_spread = inst.scatter@1(palm_crown, count=5, width=34, depth=18, min_spacing=6, yaw_jitter=3.14, scale_jitter=0.25, seed=9)
palm_trunks = inst.drop@1(trunks_spread, island, offset=-0.15, align=0, outside="remove")
palm_crowns = inst.drop@1(crowns_spread, island, offset=-0.15, align=0, outside="remove")
output palm_trunks = palm_trunks
output palm_crowns = palm_crowns

# ---- Loose rocks and driftwood-coloured boulders on the beach ----
pebble = solid.sphere@1(radius=0.5, segments=14)
pebble_rough = xform.noise@1(pebble, amplitude=0.08, scale=1.2, seed=31, tolerance=0.1)
pebble_rock = solid.texture@1(pebble_rough, tile=1.5, program="s = fbm(scale_x=4, scale_y=4); f = fbm(scale_x=24, scale_y=24); albedo = colorize(s, gradient=[0:#6a645c, 0.5:#8a8379, 1:#aaa397]); roughness = colorize(f, gradient=[0:#b8b8b8, 1:#e0e0e0]); height = colorize(f, gradient=[0:#808080, 1:#9a9a9a])")
pebble_toon = solid.toon@1(pebble_rock, bands=2, feather=0.07, shade_color=[0.68, 0.6, 0.8], outline=1.5, outline_color=[0.2, 0.15, 0.2])
pebbles_spread = inst.scatter@1(pebble_toon, count=24, width=40, depth=26, min_spacing=1.5, yaw_jitter=3.14, scale_jitter=0.6, seed=3)
pebbles = inst.drop@1(pebbles_spread, island, offset=-0.1, align=0.5, outside="remove")
output pebbles = pebbles
