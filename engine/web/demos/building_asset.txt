# Keep the material groups separate so the asset can be reused and re-skinned in graph programs.
sky = scene.sky@1(sun_elevation=31, sun_azimuth=135, sun_intensity=3.2, turbidity=3, ground_color=[0.22, 0.25, 0.29], clouds=false, seed=7)
