# Physics playground. Press Play (bottom of the page): everything falls, bounces and stacks.
# Drag a prop to pick it up and let go to throw it. Stop puts everything back.
floor_box = solid.box@1(size=[14, 0.2, 14])
floor_down = xform.move@1(floor_box, by=[0, -0.1, 0])
floor_grey = solid.material@1(floor_down, color=[0.45, 0.5, 0.45], roughness=0.9)
floor = phys.body@1(floor_grey, kind="static", shape="box")
output ground = floor

# Low walls so nothing rolls off the edge.
wall_long = solid.box@1(size=[14.4, 0.6, 0.2])

wall_n = phys.body@1(wall_n_at, kind="static", shape="box")
output wall_north = wall_n
wall_s_at = xform.move@1(wall_long, by=[0, 0.3, 7.1])
wall_s = phys.body@1(wall_s_at, kind="static", shape="box")
output wall_south = wall_s
wall_e_at = xform.move@1(wall_short, by=[7.1, 0.3, 0])
wall_e = phys.body@1(wall_e_at, kind="static", shape="box")
output wall_east = wall_e
wall_w_at = xform.move@1(wall_short, by=[-7.1, 0.3, 0])
wall_w = phys.body@1(wall_w_at, kind="static", shape="box")
output wall_west = wall_w

# Three bouncy balls dropped from different heights.

# A ramp with a ball at the top that rolls down it.
board = solid.box@1(size=[3, 0.1, 1])
board_tilt = xform.rotate@1(board, angles=[0, 0, -20])
board_at = xform.move@1(board_tilt, by=[-3, 0.6, -2])
board_wood = solid.material@1(board_at, color=[0.55, 0.4, 0.25], roughness=0.8)
ramp = phys.body@1(board_wood, kind="static", shape="convex")
output ramp = ramp
roller = solid.sphere@1(radius=0.2, segments=24)
roller_green = solid.material@1(roller, color=[0.1, 0.7, 0.3], roughness=0.4)
roller_at = xform.move@1(roller_green, by=[-4.2, 1.5, -2])
rolling = phys.body@1(roller_at, shape="sphere", angular_damping=0.02)
output roller = rolling

# A heavy block to knock things with (density 8 against 1 for the rest).
