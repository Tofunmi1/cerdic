def box_part(tag,sx,sy,sz,x,y,z,r,g,b,rough)
  sh_{tag} = solid.box@1(size=[{sx},{sy},{sz}])
  at_{tag} = xform.move@1(sh_{tag},by=[{x},{y},{z}])
  mat_{tag} = solid.material@1(at_{tag},color=[{r},{g},{b}],roughness={rough})
  output {tag} = mat_{tag}
end
