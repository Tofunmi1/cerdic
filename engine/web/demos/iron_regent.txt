# The cuirass: a shaped, beveled shell with a raised sternum and tailored pectoral plates.
chest_profile = profile.polygon@1(points=[[-0.25, 0.20], [-0.29, 0.31], [-0.25, 0.40], [-0.16, 0.445], [-0.075, 0.425], [0, 0.39], [0.075, 0.425], [0.16, 0.445], [0.25, 0.40], [0.29, 0.31], [0.25, 0.19], [0.17, 0.06], [0.08, -0.075], [0, -0.12], [-0.08, -0.075], [-0.17, 0.06]])
chest_ex = solid.extrude@1(chest_profile, height=0.105)
chest_face = xform.rotate@1(chest_ex, angles=[90, 0, 0])
chest_round = solid.round@1(chest_face, radius=0.024, segments=8)
chest_at = xform.move@1(chest_round, by=[0, 1.205, 0.155])
chest_mesh = poly.from_solid@1(chest_at)
chest_edges = poly.select@1(chest_mesh, element="edge", by="all")
chest_refined = poly.subdivide_edges@1(chest_mesh, chest_edges, cuts=2)
chest_sculpt = poly.sculpt@1(chest_refined, symmetry="x", strokes="draw 0 1.47 0.247 0.105 0.008; draw 0.13 1.43 0.25 0.09 0.006; draw -0.13 1.43 0.25 0.09 0.006; smooth 0 1.39 0.25 0.16 0.16")
chest_solid = poly.to_solid@1(chest_sculpt)
steel_base = solid.material@1(chest_solid, color=[0.34, 0.39, 0.43], metallic=0.82, roughness=0.37)
steel_finish = solid.finish@1(steel_base, clearcoat=0.18, clearcoat_roughness=0.34)
output sculpted_cuirass = steel_finish
