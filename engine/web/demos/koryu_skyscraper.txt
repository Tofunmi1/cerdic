# Silver diagrid: paired swept members wrap the main shaft and both setbacks, giving the long elevations a readable structural rhythm.
let brace_code = "function tube(ax,ay,az,bx,by,bz,r,n){ let dx=bx-ax,dy=by-ay,dz=bz-az,L=Math.hypot(dx,dy,dz)||1; dx/=L; dy/=L; dz/=L; let hx=0,hy=1,hz=0; if(Math.abs(dy)>0.9){hx=1;hy=0;} let ux=dy*hz-dz*hy,uy=dz*hx-dx*hz,uz=dx*hy-dy*hx,U=Math.hypot(ux,uy,uz)||1; ux/=U;uy/=U;uz/=U; const vx=uy*dz-uz*dy,vy=uz*dx-ux*dz,vz=ux*dy-uy*dx,A=[],B=[]; for(let i=0;i<n;i++){const t=i*6.283185/n,c=Math.cos(t),s=Math.sin(t),ox=ux*c+vx*s,oy=uy*c+vy*s,oz=uz*c+vz*s;A.push(m.vertex(ax+ox*r,ay+oy*r,az+oz*r));B.push(m.vertex(bx+ox*r,by+oy*r,bz+oz*r));} for(let i=0;i<n;i++){const j=(i+1)%n;m.face(A[i],B[i],B[j],A[j]);} m.face(...A.slice().reverse());m.face(...B);}; function wrap(x0,x1,z0,z1,y0,y1,bays,rows){const P=(side,u,y)=>side===0?[x0+(x1-x0)*u,y,z0-0.48]:side===1?[x1+0.48,y,z0+(z1-z0)*u]:side===2?[x1-(x1-x0)*u,y,z1+0.48]:[x0-0.48,y,z1-(z1-z0)*u]; for(let side=0;side<4;side++)for(let j=0;j<rows;j++)for(let i=0;i<bays;i++){const a=P(side,i/bays,y0+(y1-y0)*j/rows),b=P(side,(i+1)/bays,y0+(y1-y0)*(j+1)/rows),c=P(side,(i+1)/bays,y0+(y1-y0)*j/rows),d=P(side,i/bays,y0+(y1-y0)*(j+1)/rows);tube(...a,...b,0.105,7);tube(...c,...d,0.105,7);}} const m=new Mesh(); wrap(-21,21,-17,17,18,123,6,6); wrap(-17,17,-14,14,124,162,5,3); wrap(-12.5,12.5,-10.5,10.5,163,193,4,2); m"
